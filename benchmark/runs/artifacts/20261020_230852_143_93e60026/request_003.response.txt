          <line id="vGuide" x1="55" y1="42" x2="55" y2="252" stroke="#7ca8f7" stroke-width="1.4" stroke-dasharray="4 4"/>

          <circle id="point" cx="55" cy="42" r="5.5" fill="#2563eb" stroke="#ffffff" stroke-width="2.5"/>

          <text id="areaLabel1" x="128" y="190" text-anchor="middle" class="svg-label svg-blue" style="display:none">60 m</text>

            <text x="45" y="256" text-anchor="end">0</text>
            <text x="45" y="186" text-anchor="end">12</text>

          <text x="14" y="150" transform="rotate(-90 14 150)" text-anchor="middle" class="axis-title">v / (m·s⁻¹)</text>
          <text x="315" y="58" text-anchor="middle" class="svg-small svg-muted">斜率 = −6 m/s²</text>
        </svg>

        <div class="readouts" aria-label="当前读数">
          <div class="readout"><span>t</span><strong id="readT">0.0</strong><small>s</small></div>
          <div class="readout"><span>v</span><strong id="readV">36.0</strong><small>m/s</small></div>
          <div class="readout"><span>滑行</span><strong id="readX">0.0</strong><small>m</small></div>
        </div>
        <div class="last-readout" id="lastReadout"></div>

      <section class="card formula-card" id="formulaCard" aria-live="polite"></section>

<script>
  (function(){
    "use strict";

    const V0 = 36;
    const A = -6;
    const T_STOP = 6;
    const TOTAL_X = 108;
    const LAST_START_T = 2;
    const LAST_START_X = 60;
    const LAST_DISP = 48;

    const SCENE_X0 = 100;
    const SCENE_X1 = 900;
    const SCENE_Y = 243;
    const SCENE_SCALE = (SCENE_X1 - SCENE_X0) / TOTAL_X;

    const TIME_X0 = 100;
    const TIME_X1 = 900;
    const TIME_SCALE = (TIME_X1 - TIME_X0) / T_STOP;

    const CH = { left:55, right:410, top:42, bottom:252 };

    const state = {
      t: 0,
      step: 0,
      showLast: false,
      playing: false,
      animStart: 0,
      animFrom: 0,
      animTo: 0,
      animDur: 0
    };

    const $ = (id) => document.getElementById(id);

    const els = {
      ans1: $("ans1"),
      ans2: $("ans2"),
      statusPill: $("statusPill"),
      scenePlane: $("planeGroup"),
      velArrow: $("velArrow"),
      velLabel: $("velLabel"),
      accGroup: $("accGroup"),
      lastZoneFull: $("lastZoneFull"),
      lastZoneDone: $("lastZoneDone"),
      lastStartTick: $("lastStartTick"),
      stopLabel: $("stopLabel"),
      lastSceneBadge: $("lastSceneBadge"),
      lastSceneBadgeText: $("lastSceneBadgeText"),
      timeMarker: $("timeMarker"),
      timeDot: $("timeDot"),
      lastTimeZone: $("lastTimeZone"),
      lastBracket: $("lastBracket"),
      timeTicks: $("timeTicks"),
      timeSlider: $("timeSlider"),
      readT: $("readT"),
      readV: $("readV"),
      readX: $("readX"),
      lastReadout: $("lastReadout"),
      areaMain: $("areaMain"),
      areaLast: $("areaLast"),
      lastRegion: $("lastRegion"),
      t2Mark: $("t2Mark"),
      stopMark: $("stopMark"),
      vGuide: $("vGuide"),
      hGuide: $("hGuide"),
      point: $("point"),
      pointLabel: $("pointLabel"),
      areaLabel1: $("areaLabel1"),
      areaLabel2: $("areaLabel2"),
      areaLabelTotal: $("areaLabelTotal"),
      formulaCard: $("formulaCard"),
      stepBtns: document.querySelectorAll(".step-btn"),
      playBtn: $("playBtn"),
      resetBtn: $("resetBtn")
    };

    function posAt(t){
      return V0 * t + 0.5 * A * t * t;
    }
    function velAt(t){
      return Math.max(0, V0 + A * t);
    }
    function sceneX(t){
      return SCENE_X0 + posAt(t) * SCENE_SCALE;
    }
    function timeX(t){
      return TIME_X0 + t * TIME_SCALE;
    }
    function chartX(t){
      return CH.left + (t / T_STOP) * (CH.right - CH.left);
    }
    function chartY(v){
      return CH.bottom - (v / V0) * (CH.bottom - CH.top);
    }
    function clampT(t){
      return Math.max(0, Math.min(T_STOP, t));
    }

    function buildTimeTicks(){
      let html = "";
      for(let i=0;i<=6;i++){
        const x = timeX(i);
        const major = (i === 0 || i === 2 || i === 4 || i === 6);
        html += `<line x1="${x}" y1="${major ? 26 : 29}" x2="${x}" y2="${major ? 42 : 39}" stroke="#9aa9bd" stroke-width="${major ? 2 : 1.2}"/>`;
        html += `<text x="${x}" y="${major ? 62 : 58}" text-anchor="middle" class="${major ? "svg-label" : "svg-small"}">${i}</text>`;
      }
      els.timeTicks.innerHTML = html;
    }

    function updateStepButtons(){
      els.stepBtns.forEach(btn => {
        const s = Number(btn.dataset.step);
        btn.classList.toggle("active", s === state.step);
      });
    }

    function updateAnswers(){
      const firstKnown = state.step >= 2;
      const secondKnown = state.step >= 4;
      els.ans1.textContent = firstKnown ? "108 m" : "滑行距离 ?";
      els.ans1.classList.toggle("revealed", firstKnown);
      els.ans2.textContent = secondKnown ? "48 m" : "最后 4 s 位移 ?";
      els.ans2.classList.toggle("revealed", secondKnown);
    }

    function renderFormula(){
      let html = "";
      if(state.step === 0){
        html = `
          <div class="formula-head">
            <span class="step-tag">观察</span>
            <h3>把题意变成运动图</h3>
          </div>
          <div class="eq-grid">
            <div class="eq-row"><span class="eq-label">初速度</span><span class="eq blue-text">v₀ = 36 m/s</span></div>
            <div class="eq-row"><span class="eq-label">加速度</span><span class="eq red-text">a = −6 m/s²</span><span class="muted">与运动反向</span></div>
            <div class="eq-row"><span class="eq-label">末速度</span><span class="eq">v末 = 0</span></div>
          </div>
          <p class="hint">蓝色箭头表示速度，红色小箭头表示减速度。速度均匀变小，飞机最终停下。</p>
        `;
      } else if(state.step === 1){
        html = `
          <div class="formula-head">
            <span class="step-tag">第 1 步</span>
            <h3>先求停稳所需时间</h3>
          </div>
          <div class="formula-line">v = v₀ + at</div>
          <div class="formula-line">0 = 36 − 6t</div>
          <div class="answer-chip">t停 = <strong>6 s</strong></div>
          <p class="hint">看 v-t 图：直线与横轴相交的时刻，就是速度减到 0、飞机停稳的时刻。</p>
        `;
      } else if(state.step === 2){
        html = `
          <div class="formula-head">
            <span class="step-tag">第 (1) 问</span>
            <h3>全程滑行距离</h3>
          </div>
          <div class="formula-line">位移 = v-t 图线下面积</div>
          <div class="formula-line">s = ½ × 36 × 6</div>
          <div class="answer-chip">s = <strong>108 m</strong></div>
          <p class="hint">蓝色三角形的面积就是全程滑行距离。拖动时间滑块，可以看到面积从 0 逐渐长满整个三角形。</p>
          <button class="demo-btn" data-demo="0,6">回放全程滑行</button>
        `;
      } else if(state.step === 3){
        html = `
          <div class="formula-head">
            <span class="step-tag">第 3 步</span>
            <h3>最后 4 s 从哪个时刻开始？</h3>
          </div>
          <div class="formula-line">t起 = 6 − 4 = <strong>2 s</strong></div>
          <div class="formula-line">v(2) = 36 − 6×2 = <strong>24 m/s</strong></div>
          <div class="formula-line" style="font-size:15px">x(2) = 36×2 − ½×6×2² = 60 m</div>
          <p class="hint">黄色区域就是最后 4 s。它在跑道上从 60 m 处开始，一直延伸到 108 m 处的停止点。</p>
          <button class="demo-btn" data-demo="2,6">扫过最后 4 s</button>
        `;
      } else {
        html = `
          <div class="formula-head">
            <span class="step-tag">第 (2) 问</span>
            <h3>最后 4 s 的位移</h3>
          </div>
          <div class="formula-line">s后4 = ½ × (24 + 0) × 4</div>
          <div class="answer-chip amber-chip">s后4 = <strong>48 m</strong></div>
          <p class="hint">核对：前 2 s 滑行 60 m，最后 4 s 滑行 48 m，60 + 48 = 108 m，正好是全程距离。</p>
          <button class="demo-btn" data-demo="2,6">回放最后 4 s</button>
        `;
      }
      els.formulaCard.innerHTML = html;
    }

    function updateAreas(t){
      const v = velAt(t);
      const mainEnd = state.showLast ? Math.min(t, LAST_START_T) : t;
      const mainPoints = underCurvePoints(0, mainEnd);
      if(mainPoints){
        els.areaMain.setAttribute("points", mainPoints);
        els.areaMain.style.display = "";
      } else {
        els.areaMain.style.display = "none";
      }

      if(state.showLast){
        els.lastRegion.style.display = "";
        els.t2Mark.style.display = "";
      } else {
        els.lastRegion.style.display = "none";
        els.t2Mark.style.display = "none";
      }

      if(state.showLast && t > LAST_START_T + 0.001){
        const lastPoints = underCurvePoints(LAST_START_T, t);
        els.areaLast.setAttribute("points", lastPoints);
        els.areaLast.style.display = "";
      } else {
        els.areaLast.style.display = "none";
      }

      els.stopMark.style.display = (state.step >= 1 || t >= 5.995) ? "" : "none";

      const showTotalLabel = state.step === 2 && t >= 5.995;
      els.areaLabelTotal.style.display = showTotalLabel ? "" : "none";

      const showFirstPart = state.showLast && t >= LAST_START_T - 0.001;
      els.areaLabel1.style.display = showFirstPart ? "" : "none";

      const showLastPart = state.step === 4 && t >= 5.995;
      els.areaLabel2.style.display = showLastPart ? "" : "none";
    }

    function underCurvePoints(t1, t2){
      if(t2 <= t1 + 0.0001) return "";
      const x1 = chartX(t1);
      const x2 = chartX(t2);
      const y1 = chartY(velAt(t1));
      const y2 = chartY(velAt(t2));
      return `${x1},${CH.bottom} ${x1},${y1} ${x2},${y2} ${x2},${CH.bottom}`;
    }

    function updateChart(t){
      const v = velAt(t);
      const px = chartX(t);
      const py = chartY(v);

      els.point.setAttribute("cx", px);
      els.point.setAttribute("cy", py);

      els.vGuide.setAttribute("x1", px);
      els.vGuide.setAttribute("x2", px);
      els.vGuide.setAttribute("y1", py);
      els.vGuide.setAttribute("y2", CH.bottom);

      els.hGuide.setAttribute("x1", CH.left);
      els.hGuide.setAttribute("x2", px);
      els.hGuide.setAttribute("y1", py);
      els.hGuide.setAttribute("y2", py);

      let lx = px + 10;
      let ly = py - 10;
      let anchor = "start";
      if(px > 340){
        lx = px - 10;
        anchor = "end";
      }
      if(py < 62){
        ly = py + 22;
      }
      els.pointLabel.setAttribute("x", lx);
      els.pointLabel.setAttribute("y", ly);
      els.pointLabel.setAttribute("text-anchor", anchor);
      els.pointLabel.textContent = `v=${v.toFixed(1)}`;

      updateAreas(t);
    }

    function updateScene(t){
      const v = velAt(t);
      const x = posAt(t);
      const planeX = sceneX(t);

      els.scenePlane.setAttribute("transform", `translate(${planeX} ${SCENE_Y})`);

      const arrowStart = planeX - 18;
      if(v > 0.05){
        const arrowLen = Math.max(5, v * 2.25);
        els.velArrow.style.display = "";
        els.velArrow.setAttribute("x1", arrowStart);
        els.velArrow.setAttribute("x2", arrowStart + arrowLen);
        els.velLabel.style.display = "";
        els.velLabel.setAttribute("x", arrowStart);
        els.velLabel.setAttribute("text-anchor", "start");
        els.velLabel.textContent = `v = ${v.toFixed(1)} m/s`;
        els.accGroup.style.display = "";
        els.accGroup.setAttribute("transform", `translate(${planeX - 18} 180)`);
      } else {
        els.velArrow.style.display = "none";
        els.velLabel.style.display = "";
        els.velLabel.setAttribute("x", planeX - 8);
        els.velLabel.setAttribute("text-anchor", "end");
        els.velLabel.textContent = "v = 0（已停稳）";
        els.accGroup.style.display = "none";
      }

      els.stopLabel.textContent = state.step >= 2 ? "108 m" : "终点 ?";

      if(state.showLast){
        const lastStartSceneX = SCENE_X0 + LAST_START_X * SCENE_SCALE;
        const stopX = SCENE_X1;
        els.lastZoneFull.style.display = "";
        els.lastZoneFull.setAttribute("x", lastStartSceneX);
        els.lastZoneFull.setAttribute("width", stopX - lastStartSceneX);

        const doneWidth = Math.max(0, Math.min(planeX, stopX) - lastStartSceneX);
        els.lastZoneDone.style.display = "";
        els.lastZoneDone.setAttribute("x", lastStartSceneX);
        els.lastZoneDone.setAttribute("width", doneWidth);

        els.lastStartTick.style.display = "";
        els.lastStartTick.setAttribute("transform", `translate(${lastStartSceneX} 0)`);

        els.lastSceneBadge.style.display = "";
        if(state.step === 4 && t >= 5.995){
          els.lastSceneBadgeText.textContent = "最后 4 s：48 m";
        } else {
          els.lastSceneBadgeText.textContent = "最后 4 s 区间";
        }
      } else {
        els.lastZoneFull.style.display = "none";
        els.lastZoneDone.style.display = "none";
        els.lastStartTick.style.display = "none";
        els.lastSceneBadge.style.display = "none";
      }
    }

    function updateTimeline(t){
      const tx = timeX(t);
      els.timeMarker.setAttribute("x1", tx);
      els.timeMarker.setAttribute("x2", tx);
      els.timeDot.setAttribute("cx", tx);

      if(state.showLast){
        els.lastTimeZone.style.display = "";
        els.lastBracket.style.display = "";
      } else {
        els.lastTimeZone.style.display = "none";
        els.lastBracket.style.display = "none";
      }
    }

    function updateReadouts(t){
      const v = velAt(t);
      const x = posAt(t);
      els.readT.textContent = t.toFixed(1);
      els.readV.textContent = v.toFixed(1);
      els.readX.textContent = x.toFixed(1);
      els.statusPill.textContent = v <= 0.01 ? "已停稳" : `t = ${t.toFixed(1)} s`;

      if(state.showLast){
        const within = Math.max(0, x - LAST_START_X);
        const isFinal = state.step === 4 && t >= 5.995;
        els.lastReadout.style.display = "";
        els.lastReadout.classList.toggle("final", isFinal);
        if(isFinal){
          els.lastReadout.innerHTML = `最后 4 s 内位移：<strong>${within.toFixed(1)} m</strong>`;
        } else {
          els.lastReadout.innerHTML = `最后 4 s 区间内已滑行：<strong>${within.toFixed(1)} m</strong>`;
        }
      } else {
        els.lastReadout.style.display = "none";
        els.lastReadout.innerHTML = "";
      }
    }

    function renderAll(){
      const t = state.t;
      els.timeSlider.value = String(t);
      updateScene(t);
      updateTimeline(t);
      updateChart(t);
      updateReadouts(t);
    }

    function setStep(step){
      state.playing = false;
      state.step = step;
      state.showLast = step >= 3;
      const targets = [0, 0, 6, 2, 6];
      state.t = targets[step] ?? 0;
      updateStepButtons();
      updateAnswers();
      renderFormula();
      updatePlayButton();
      renderAll();
    }

    function updatePlayButton(){
      els.playBtn.textContent = state.playing ? "暂停" : "播放";
      els.playBtn.classList.toggle("primary", !state.playing);
    }

    function playTo(endT, duration){
      endT = clampT(endT);
      const startT = state.t;
      if(Math.abs(endT - startT) < 0.001){
        state.playing = false;
        updatePlayButton();
        return;
      }
      state.playing = true;
      state.animStart = performance.now();
      state.animFrom = startT;
      state.animTo = endT;
      state.animDur = duration || Math.max(650, Math.abs(endT - startT) * 380);
      updatePlayButton();
      requestAnimationFrame(animTick);
    }

    function animTick(now){
      if(!state.playing) return;
      const p = Math.min(1, (now - state.animStart) / state.animDur);
      state.t = state.animFrom + (state.animTo - state.animFrom) * p;
      if(p >= 1){
        state.t = state.animTo;
        state.playing = false;
        updatePlayButton();
      }
      renderAll();
      if(state.playing){
        requestAnimationFrame(animTick);
      }
    }

    function togglePlay(){
      if(state.playing){
        state.playing = false;
        updatePlayButton();
        return;
      }
      if(state.t >= 5.995){
        state.t = 0;
      }
      playTo(T_STOP, Math.max(700, (T_STOP - state.t) * 380));
    }

    // Initialize timeline ticks
    buildTimeTicks();

    // Step buttons
    els.stepBtns.forEach(btn => {
      btn.addEventListener("click", () => setStep(Number(btn.dataset.step)));
    });

    // Slider
    els.timeSlider.addEventListener("input", (e) => {
      state.playing = false;
      state.t = clampT(parseFloat(e.target.value));
      updatePlayButton();
      renderAll();
    });

    // Play / reset
    els.playBtn.addEventListener("click", togglePlay);
    els.resetBtn.addEventListener("click", () => setStep(0));

    // Demo buttons inside formula card
    els.formulaCard.addEventListener("click", (e) => {
      const btn = e.target.closest("[data-demo]");
      if(!btn) return;
      const parts = btn.dataset.demo.split(",").map(Number);
      state.playing = false;
      state.t = clampT(parts[0]);
      renderAll();
      playTo(parts[1], Math.max(750, Math.abs(parts[1] - parts[0]) * 380));
    });

    // First render
    setStep(0);
  })();
</script>
</body>
