  function stopPlay() {
    playing = false;
    if (raf) {
      cancelAnimationFrame(raf);
      raf = null;
    }
    $("playBtn").textContent = "播放";
  }

  function startPlay() {
    if (playing) return;
    playing = true;
    dwell = 0;
    $("playBtn").textContent = "暂停";

    if (state === 4) {
      animDir = scanY < -13 ? 1 : -1;
    } else {
      animDir = currentX > 1 ? -1 : 1;
    }

    let last = performance.now();

    function frame(ts) {
      if (!playing) return;
      const dt = Math.min(50, ts - last) / 1000;
      last = ts;

      if (dwell > 0) {
        dwell -= dt;
      } else if (state === 4) {
        const prev = scanY;
        scanY += animDir * 4.2 * dt;

        if (prev < -13 && scanY >= -13 && animDir > 0) {
          scanY = -13;
          dwell = 0.38;
        } else if (prev > -13 && scanY <= -13 && animDir < 0) {
          scanY = -13;
          dwell = 0.38;
        }

        if (scanY >= -8) {
          scanY = -8;
          animDir = -1;
          dwell = 0.18;
        } else if (scanY <= -30) {
          scanY = -30;
          animDir = 1;
          dwell = 0.18;
        }
      } else {
        currentX += animDir * 1.9 * dt;
        if (currentX >= XMAX) {
          currentX = XMAX;
          animDir = -1;
          dwell = 0.16;
        } else if (currentX <= XMIN) {
          currentX = XMIN;
          animDir = 1;
          dwell = 0.16;
        }
      }

      renderGraph();
      raf = requestAnimationFrame(frame);
    }

    raf = requestAnimationFrame(frame);
  }

  function setState(n) {
    if (n < 1 || n > 4) return;
    stopPlay();
    state = n;
    renderAll();
  }

  function clientToSvgPoint(evt) {
    const pt = svg.createSVGPoint();
    pt.x = evt.clientX;
    pt.y = evt.clientY;
    return pt.matrixTransform(svg.getScreenCTM().inverse());
  }

  function movePointFromClient(evt) {
    if (state === 4) return;
    const p = clientToSvgPoint(evt);
    let x = XMIN + (p.x - L) / (R - L) * (XMAX - XMIN);
    x = Math.max(XMIN, Math.min(XMAX, x));
    currentX = Math.round(x * 100) / 100;
    renderGraph();
  }

    e.target.setPointerCapture(e.pointerId);
    movePointFromClient(e);
    e.preventDefault();
  });

  $("pointHit").addEventListener("pointermove", e => {
    if (!draggingPoint) return;
    movePointFromClient(e);
    e.preventDefault();
  });

  function endPointDrag(e) {
    draggingPoint = false;
    if (e.target.hasPointerCapture && e.target.hasPointerCapture(e.pointerId)) {
      e.target.releasePointerCapture(e.pointerId);
    }
  }

  $("pointHit").addEventListener("pointerup", endPointDrag);
  $("pointHit").addEventListener("pointercancel", endPointDrag);
