<script>
(function(){
  "use strict";

  /* ================= 物理常量 ================= */
  const V0 = 36;        // m/s
  const A  = 6;         // m/s^2 大小
  const TSTOP = 6;      // s
  const STOT  = 108;    // m
  const TLAST = 2;      // 最后 4 s 的起点时刻
  const SLAST = 60;     // s(2s)=60 m

  const FONT = 'system-ui,-apple-system,"Segoe UI","PingFang SC","Microsoft YaHei",sans-serif';

  function sOf(t){ return V0*t - 0.5*A*t*t; }
  function vOf(t){ return Math.max(0, V0 - A*t); }
  function clamp(x,a,b){ return Math.min(b, Math.max(a,x)); }

  /* ================= 状态 ================= */
  const state = {
    t: 0,
    step: 0,
    playing: false,
    showStop: false,
    showArea: false,
    showLast4: false,
    showReverse: false
  };

  /* ================= 步骤定义 ================= */
  const STEPS = [
    {
      btn: '① 看清题设',
      text: '飞机以 36 m/s 向右滑行，加速度与速度反向，速度每秒减少 6 m/s。',
      formula: 'v₀ = 36 m/s ，a = −6 m/s² ，v末 = 0'
    },
    {
      btn: '② 先求停止时间',
      text: '每秒少 6 m/s，从 36 m/s 减到 0 一共要 6 s——这是整段滑行的总时间。',
      formula: 't₀ = v₀ / |a| = 36 ÷ 6 = 6 s'
    },
    {
      btn: '③ 总距离 = 图像面积',
      text: 'v–t 图线与 t 轴围成的三角形面积，就是滑行的总距离。底 6 s，高 36 m/s。',
      formula: 's = ½ · 6 s · 36 m/s = 108 m'
    },
    {
      btn: '④ 最后 4 s 是哪一段',
      text: '总时间 6 s，最后 4 s 从 t = 2 s 开始。此刻速度还剩 24 m/s，再减到 0。',
      formula: 'v(2 s) = 24 m/s → s₄ = (24+0)/2 × 4 = 48 m'
    },
    {
      btn: '⑤ 倒过来看更简单',
      text: '把这段运动倒放：从静止出发，以 6 m/s² 加速 4 s，位移完全一样。',
      formula: 's₄ = ½ · 6 · 4² = 48 m'
    }
  ];

  /* ================= 元素 ================= */
  const sceneCv = document.getElementById('scene');
  const chartCv = document.getElementById('chart');
  const sctx = sceneCv.getContext('2d');
  const cctx = chartCv.getContext('2d');
  const slider = document.getElementById('timeSlider');
  const playBtn = document.getElementById('playBtn');
  const resetBtn = document.getElementById('resetBtn');
  const timeLabel = document.getElementById('timeLabel');
  const stepText = document.getElementById('stepText');
  const stepFormula = document.getElementById('stepFormula');
  const stepButtons = document.getElementById('stepButtons');

  /* ================= 画布尺寸 ================= */
  function dpr(){ return Math.min(window.devicePixelRatio || 1, 2); }

  function sizeCanvas(cv, ctx){
    const r = cv.getBoundingClientRect();
    const w = Math.max(20, Math.round(r.width));
    const h = Math.max(20, Math.round(r.height));
    const d = dpr();
    const pw = Math.round(w*d), ph = Math.round(h*d);
    if(cv.width !== pw || cv.height !== ph){
      cv.width = pw;
      cv.height = ph;
    }
    ctx.setTransform(d,0,0,d,0,0);
    return {w:w, h:h};
  }

  function roundRect(ctx, x, y, w, h, r){
    r = Math.min(r, w/2, h/2);
    ctx.beginPath();
    ctx.moveTo(x+r, y);
    ctx.arcTo(x+w, y, x+w, y+h, r);
    ctx.arcTo(x+w, y+h, x, y+h, r);
    ctx.arcTo(x, y+h, x, y, r);
    ctx.arcTo(x, y, x+w, y, r);
    ctx.closePath();
  }

  function arrow(ctx, ax, ay, bx, by, color, lw){
    const ang = Math.atan2(by-ay, bx-ax);
    const head = 9;
    ctx.strokeStyle = color;
    ctx.fillStyle = color;
    ctx.lineWidth = lw;
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(ax, ay);
    ctx.lineTo(bx - Math.cos(ang)*head*0.85, by - Math.sin(ang)*head*0.85);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(bx, by);
    ctx.lineTo(bx - Math.cos(ang-0.42)*head, by - Math.sin(ang-0.42)*head);
    ctx.lineTo(bx - Math.cos(ang+0.42)*head, by - Math.sin(ang+0.42)*head);
    ctx.closePath();
    ctx.fill();
  }

  /* ================= 飞机图形 ================= */
  function drawPlane(ctx, x, y, k){
    ctx.save();
    ctx.translate(x, y);
    ctx.scale(k, k);

    // 尾翼
    ctx.fillStyle = '#334155';
    ctx.beginPath();
    ctx.moveTo(-50, 0);
    ctx.lineTo(-62, -22);
    ctx.lineTo(-40, -1);
    ctx.closePath();
    ctx.fill();

    // 机翼
    ctx.fillStyle = '#475569';
    ctx.beginPath();
    ctx.moveTo(-30, 1);
    ctx.lineTo(-46, 20);
    ctx.lineTo(-8, 3);
    ctx.closePath();
    ctx.fill();

    // 机身
    ctx.fillStyle = '#1e293b';
    ctx.beginPath();
    ctx.moveTo(3, 0);
    ctx.quadraticCurveTo(-6, -9, -26, -9);
    ctx.lineTo(-50, -6.5);
    ctx.quadraticCurveTo(-61, 0, -50, 6.5);
    ctx.lineTo(-26, 9);
    ctx.quadraticCurveTo(-6, 9, 3, 0);
    ctx.closePath();
    ctx.fill();

    // 舷窗
    ctx.fillStyle = '#93c5fd';
    ctx.beginPath();
    ctx.ellipse(-4, -1, 8.5, 3, 0, 0, Math.PI*2);
    ctx.fill();

    ctx.restore();
  }

  /* ================= 主图：滑动场景 ================= */
  function drawScene(w, h){
    const ctx = sctx;
    ctx.clearRect(0, 0, w, h);

    const t = state.t;
    const sNow = sOf(t);
    const vNow = vOf(t);

    const padL = 84, padR = 30;
    const x0 = padL, x1 = w - padR;
    const scale = (x1 - x0) / STOT;
    const X = m => x0 + m * scale;

    const rwTop = h - 132;
    const rwBot = h - 58;
    const rwMid = (rwTop + rwBot) / 2;

    /* --- 跑道底色 --- */
    const g = ctx.createLinearGradient(0, rwTop, 0, rwBot);
    g.addColorStop(0, '#e9eff8');
    g.addColorStop(1, '#ccd8e8');
    roundRect(ctx, x0 - 14, rwTop, (x1 + 14) - (x0 - 14), rwBot - rwTop, 9);
    ctx.fillStyle = g;
    ctx.fill();

    /* --- 最后 4 s 的高亮区 --- */
    if(state.showLast4){
      ctx.save();
      roundRect(ctx, x0 - 14, rwTop, (x1 + 14) - (x0 - 14), rwBot - rwTop, 9);
      ctx.clip();
      ctx.fillStyle = 'rgba(245,158,11,0.26)';
      ctx.fillRect(X(SLAST), rwTop, X(STOT) - X(SLAST), rwBot - rwTop);
      ctx.restore();

      ctx.strokeStyle = '#f59e0b';
      ctx.lineWidth = 2;
      ctx.setLineDash([6,5]);
      ctx.beginPath();
      ctx.moveTo(X(SLAST), rwTop - 12);
      ctx.lineTo(X(SLAST), rwBot + 6);
      ctx.stroke();
      ctx.setLineDash([]);

      ctx.fillStyle = '#b45309';
      ctx.font = '700 12px ' + FONT;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'bottom';
      ctx.fillText('t = 2 s 从这里算起', X(SLAST), rwTop - 16);
    }

    /* --- 终点虚线 --- */
    if(state.showStop){
      ctx.strokeStyle = 'rgba(37,99,235,0.55)';
      ctx.lineWidth = 2;
      ctx.setLineDash([5,5]);
      ctx.beginPath();
      ctx.moveTo(X(STOT), rwTop - 12);
      ctx.lineTo(X(STOT), rwBot + 6);
      ctx.stroke();
      ctx.setLineDash([]);
      ctx.fillStyle = '#1d4ed8';
      ctx.font = '700 12px ' + FONT;
      ctx.textAlign = 'right';
      ctx.textBaseline = 'bottom';
      ctx.fillText('静止处', X(STOT) + 6, rwTop - 16);
    }

    /* --- 跑道中线 --- */
    ctx.save();
    roundRect(ctx, x0 - 14, rwTop, (x1 + 14) - (x0 - 14), rwBot - rwTop, 9);
    ctx.clip();
    ctx.strokeStyle = 'rgba(255,255,255,0.85)';
    ctx.lineWidth = 2;
    ctx.setLineDash([15,13]);
    ctx.beginPath();
    ctx.moveTo(x0 - 20, rwMid + 24);
    ctx.lineTo(x1 + 20, rwMid + 24);
    ctx.stroke();
    ctx.setLineDash([]);
    ctx.restore();

    /* --- 刻度 --- */
    ctx.font = '11px ' + FONT;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    [0, 36, 72, 108].forEach(function(m){
      const x = X(m);
      ctx.strokeStyle = '#9aa9bd';
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(x, rwBot);
      ctx.lineTo(x, rwBot + 7);
      ctx.stroke();
      ctx.fillStyle = '#7c8ba1';
      ctx.fillText(m + ' m', x, rwBot + 11);
    });

    /* --- 已滑行距离条 --- */
    ctx.fillStyle = 'rgba(148,163,184,0.45)';
    ctx.fillRect(x0, rwBot + 30, x1 - x0, 5);
    ctx.fillStyle = '#2563eb';
    ctx.fillRect(x0, rwBot + 30, Math.max(0, X(sNow) - x0), 5);

    /* --- 速度矢量（绿） --- */
    if(vNow > 0.4){
      const ay = rwMid - 27;
      const axA = clamp(X(sNow) + 8, x0 + 4, x1 - 6);
      const len = Math.max(16, vNow * 1.55);
      const axB = Math.min(axA + len, x1 + 6);
      arrow(ctx, axA, ay, axB, ay, '#10b981', 3);
      ctx.fillStyle = '#059669';
      ctx.font = '700 12px ' + FONT;
      ctx.textAlign = 'left';
      ctx.textBaseline = 'bottom';
      ctx.fillText('v = ' + vNow.toFixed(1) + ' m/s', axA - 2, ay - 8);
    }

    /* --- 加速度矢量（红，与 v 反向） --- */
    const aHead = clamp(X(sNow) - 12, x0 + 56, x1 + 8);
    const aTail = aHead + 46;
    const ayA = rwMid + 30;
    arrow(ctx, aTail, ayA, aHead, ayA, '#ef4444', 3);
    ctx.fillStyle = '#dc2626';
    ctx.font = '700 12px ' + FONT;
    ctx.textAlign = 'left';
    ctx.textBaseline = 'bottom';
    ctx.fillText('a = 6 m/s²', aTail + 4, ayA - 8);

    /* --- 飞机 --- */
    drawPlane(ctx, X(sNow), rwMid - 2, 0.8);

    /* --- 左下角时间 --- */
    ctx.fillStyle = '#94a3b8';
    ctx.font = '11.5px ' + FONT;
    ctx.textAlign = 'right';
    ctx.textBaseline = 'top';
    ctx.fillText('t = ' + t.toFixed(1) + ' s', w - 12, 12);
  }

  /* ================= v–t 图像 ================= */
  function drawChart(w, h){
    const ctx = cctx;
    ctx.clearRect(0, 0, w, h);

    const tMax = 7.1, vMax = 42;
    const L = 52, R = w - 20, Tp = 18, Bt = h - 42;
    const X = t => L + (t/tMax)*(R-L);
    const Y = v => Bt - (v/vMax)*(Bt-Tp);

    /* 网格 */
    ctx.strokeStyle = '#e9eff8';
    ctx.lineWidth = 1;
    for(let v = 0; v <= 36; v += 12){
      ctx.beginPath(); ctx.moveTo(L, Y(v)); ctx.lineTo(R, Y(v)); ctx.stroke();
    }
    for(let t = 0; t <= 6; t += 2){
      ctx.beginPath(); ctx.moveTo(X(t), Tp); ctx.lineTo(X(t), Bt); ctx.stroke();
    }

    /* 整个三角形面积 */
    ctx.beginPath();
    ctx.moveTo(X(0), Y(0));
    ctx.lineTo(X(0), Y(36));
    ctx.lineTo(X(6), Y(0));
    ctx.closePath();
    ctx.fillStyle = state.showArea ? 'rgba(37,99,235,0.17)' : 'rgba(37,99,235,0.09)';
    ctx.fill();

    /* 最后 4 s 的面积 */
    if(state.showLast4){
      ctx.beginPath();
      ctx.moveTo(X(2), Y(0));
      ctx.lineTo(X(2), Y(24));
      ctx.lineTo(X(6), Y(0));
      ctx.closePath();
      ctx.fillStyle = 'rgba(245,158,11,0.45)';
      ctx.fill();

  function animateTimeTo(target, dur){
    if(tweenRAF) cancelAnimationFrame(tweenRAF);
    const from = state.t;
    const t0 = performance.now();
    dur = dur || 420;
    function frame(now){
      const p = Math.min(1, (now - t0) / dur);
      const e = 1 - Math.pow(1 - p, 3);
      setTime(from + (target - from) * e);
      render();
      if(p < 1) tweenRAF = requestAnimationFrame(frame);
      else tweenRAF = null;
    }
    tweenRAF = requestAnimationFrame(frame);
  }

  let playRAF = null;
  let lastNow = 0;
  const SPEED = 1.5; // 模拟秒 / 真实秒

  function stopPlay(){
    state.playing = false;
    if(playRAF) cancelAnimationFrame(playRAF);
    playRAF = null;

  function startPlay(){
    if(tweenRAF){ cancelAnimationFrame(tweenRAF); tweenRAF = null; }
    state.playing = true;
    playBtn.textContent = '暂停';
    lastNow = performance.now();
    function loop(now){
      if(!state.playing) return;
      const dt = (now - lastNow) / 1000;
      lastNow = now;
      let t = state.t + dt * SPEED;
      if(t >= TSTOP){ t = TSTOP; }
      setTime(t);
      render();
      if(t >= TSTOP - 0.0001){ stopPlay(); return; }
      playRAF = requestAnimationFrame(loop);
    }
    playRAF = requestAnimationFrame(loop);
  }

    STEPS.forEach(function(st, i){

      b.textContent = st.btn;

  function setStep(i){
    state.step = i;
    state.showStop = i >= 1;
    state.showArea = i >= 2;
    state.showLast4 = i >= 3;
    state.showReverse = i >= 4;

    const btns = stepButtons.querySelectorAll('.step-btn');
    btns.forEach(function(b, k){ b.classList.toggle('active', k === i); });

    stepText.textContent = STEPS[i].text;
    stepFormula.textContent = STEPS[i].formula;
    stepFormula.classList.toggle('warm', i >= 3);

    if(state.playing) stopPlay();

    const targetT = [0, 6, 6, 2, 2][i];
    animateTimeTo(targetT, 480);
  }

  /* ================= 事件绑定 ================= */
  slider.addEventListener('input', function(){
    if(state.playing) stopPlay();
    if(tweenRAF){ cancelAnimationFrame(tweenRAF); tweenRAF = null; }
    setTime(parseFloat(slider.value));
    render();
  });

  playBtn.addEventListener('click', function(){
    if(state.playing){
      stopPlay();
    } else {
      if(state.t >= TSTOP - 0.01) setTime(0);
      startPlay();
    }
  });

  resetBtn.addEventListener('click', function(){
    if(state.playing) stopPlay();
    if(tweenRAF){ cancelAnimationFrame(tweenRAF); tweenRAF = null; }
    state.showStop = false;
    state.showArea = false;
    state.showLast4 = false;
    state.showReverse = false;
    state.step = 0;
    const btns = stepButtons.querySelectorAll('.step-btn');
    btns.forEach(function(b, k){ b.classList.toggle('active', k === 0); });
    stepText.textContent = STEPS[0].text;
    stepFormula.textContent = STEPS[0].formula;
    stepFormula.classList.remove('warm');
    playBtn.textContent = '播放';
    setTime(0);
    render();
  });

  let resizeTimer = null;
  window.addEventListener('resize', function(){
    if(resizeTimer) clearTimeout(resizeTimer);
    resizeTimer = setTimeout(render, 110);
  });

  /* ================= 启动 ================= */
  buildStepButtons();
  stepText.textContent = STEPS[0].text;
  stepFormula.textContent = STEPS[0].formula;
  setTime(0);
  render();

  // 首帧字体/布局稳定后再渲染一次
  setTimeout(render, 60);
  window.addEventListener('load', render);

})();
</script>

### 互动机制与操作解析

这个页面把抽象的匀减速运动拆成了“看场景”和“读图像”两条并行线索，让你在动手拖动和播放中，自己发现速度、时间、位移之间的对应关系。

**主图与实时读数**
左侧主图用跑道上的飞机位置直观呈现滑行进度，飞机上方绿色箭头表示瞬时速度，红色箭头表示与速度反向的加速度。右下角三个读数会同步更新当前时间、速度和已滑行距离，让你拖动时间滑块时能立刻看到数值变化。

**v–t 图像的面积隐喻**
右侧图像把速度随时间的变化画成一条斜线，图线与 t 轴围成的三角形面积就是滑行位移。随着时间推进，蓝色标记点沿着斜线向下移动，底部灰色区域逐渐填满，你不需要背公式就能感受到“面积=位移”的含义。

**五步拆解与自动演示**
下方的步骤按钮把解题过程拆成五步：从看清题设到求总距离，再到定位最后 4 秒。点击任意步骤，时间滑块会自动滑到对应位置，主图和 v–t 图同步高亮相关区域，你只需要观察图形变化和公式之间的联动关系。

**优化建议：** 如果想调整题目参数，可以修改脚本开头的 V0、A、TSTOP 和 STOT 四个常量，其他数值和图形会随它们自动更新。
