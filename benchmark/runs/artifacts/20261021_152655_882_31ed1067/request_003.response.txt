  function gapAt(t, mode, x0){
    if (mode === "case1") {
      const xA = t <= 8 ? 56 + 16 * t - t * t : 120;
      const xB = 4 * t + 0.5 * t * t;
      return xA - xB;
    } else {
      const xA = t <= 8 ? 16 * t - t * t : 64;
      const xB = x0 + 4 * t + 0.5 * t * t;
      return xB - xA;
    }
  }

  function rootInfo(x0){
    if (x0 < 24 - 1e-9) {
      const r = Math.sqrt(16 - (2 / 3) * x0);
      return { count: 2, roots: [4 - r, 4 + r] };
    }
    if (Math.abs(x0 - 24) < 1e-9) {
      return { count: 1, roots: [4] };
    }
    return { count: 0, roots: [] };
  }

  function niceStep(raw){
    if (!isFinite(raw) || raw <= 0) return 1;
    const pow = Math.pow(10, Math.floor(Math.log10(raw)));
    const n = raw / pow;
    let base;
    if (n <= 1) base = 1;
    else if (n <= 2) base = 2;
    else if (n <= 5) base = 5;
    else base = 10;
    return base * pow;
  }

  function labelNum(v, step){
    if (step >= 1) return String(Math.round(v));
    return v.toFixed(1);
  }

  function renderGap(s){
    const W = 460, H = 245, L = 48, R = 16, T = 22, B = 34;
    const tmax = state.mode === "case1" ? 14 : 10;

    const vals = [];
    for (let i = 0; i <= 240; i++) {
      const t = tmax * i / 240;
      vals.push(gapAt(t, state.mode, state.x0));
    }

    let mn = Math.min.apply(null, vals);
    let mx = Math.max.apply(null, vals);
    let pad = Math.max(4, (mx - mn) * 0.16);
    let rawMin = Math.min(0, mn - pad);
    let rawMax = Math.max(0, mx + pad);
    if (rawMax - rawMin < 8) {
      rawMin -= 4;
      rawMax += 4;
    }
    const step = niceStep((rawMax - rawMin) / 4);
    const ymin = Math.floor(rawMin / step) * step;
    const ymax = Math.ceil(rawMax / step) * step;

    const xp = function(t){ return L + (t / tmax) * (W - L - R); };
    const yp = function(v){ return T + (ymax - v) / (ymax - ymin) * (H - T - B); };

    let g = '<rect x="0" y="0" width="460" height="245" rx="10" fill="#ffffff"></rect>';

    for (let t = 0; t <= tmax + 1e-9; t += 2) {
      const x = xp(t);
      g += '<line x1="' + x.toFixed(1) + '" y1="' + T + '" x2="' + x.toFixed(1) + '" y2="' + (H - B) + '" stroke="' + COLORS.grid + '" stroke-width="1"></line>';
      g += '<text x="' + x.toFixed(1) + '" y="' + (H - 12) + '" text-anchor="middle" font-size="10" fill="#7d8da3">' + t + '</text>';
    }

    const vCount = Math.min(20, Math.round((ymax - ymin) / step));
    for (let i = 0; i <= vCount; i++) {
      const v = ymin + i * step;
      const y = yp(v);
      g += '<line x1="' + L + '" y1="' + y.toFixed(1) + '" x2="' + (W - R) + '" y2="' + y.toFixed(1) + '" stroke="' + COLORS.grid + '" stroke-width="1"></line>';
      g += '<text x="' + (L - 6) + '" y="' + (y + 3.5).toFixed(1) + '" text-anchor="end" font-size="10" fill="#7d8da3">' + labelNum(v, step) + '</text>';
    }

    g += '<line x1="' + L + '" y1="' + (H - B) + '" x2="' + (W - R) + '" y2="' + (H - B) + '" stroke="' + COLORS.axis + '" stroke-width="1.2"></line>';
    g += '<line x1="' + L + '" y1="' + T + '" x2="' + L + '" y2="' + (H - B) + '" stroke="' + COLORS.axis + '" stroke-width="1.2"></line>';
    g += '<text x="' + (W - R) + '" y="' + (H - 12) + '" text-anchor="end" font-size="10.5" fill="#70839b">t / s</text>';

    const y0 = yp(0);
    g += '<line x1="' + L + '" y1="' + y0.toFixed(1) + '" x2="' + (W - R) + '" y2="' + y0.toFixed(1) + '" stroke="' + COLORS.zero + '" stroke-width="1.4" opacity="0.58"></line>';
    g += '<text x="' + (W - R) + '" y="' + (y0 - 5).toFixed(1) + '" text-anchor="end" font-size="9.5" fill="#c45767">0 轴：相遇</text>';

    let pts = [];
    for (let i = 0; i <= 240; i++) {
      const t = tmax * i / 240;
      const v = gapAt(t, state.mode, state.x0);
      pts.push(xp(t).toFixed(1) + "," + yp(v).toFixed(1));
    }
    const lineColor = state.mode === "case1" ? "#355d8f" : "#4b6f97";
    g += '<polyline points="' + pts.join(" ") + '" fill="none" stroke="' + lineColor + '" stroke-width="3" stroke-linecap="round" stroke-linejoin="round"></polyline>';

    let roots = [];
    if (state.mode === "case1") {
      if (state.revealed.meet) roots = [12];
    } else {
      roots = rootInfo(state.x0).roots;
    }

    roots.forEach(function(rt, idx){
      if (rt >= 0 && rt <= tmax) {
        const x = xp(rt);
        g += '<line x1="' + x.toFixed(1) + '" y1="' + T + '" x2="' + x.toFixed(1) + '" y2="' + (H - B) + '" stroke="#df5b68" stroke-width="1.2" stroke-dasharray="3 4" opacity="0.55"></line>';
        g += '<circle cx="' + x.toFixed(1) + '" cy="' + y0.toFixed(1) + '" r="4.6" fill="#ffffff" stroke="#df5b68" stroke-width="2.2"></circle>';
        if (state.mode === "case3" || state.revealed.meet) {
          g += '<text x="' + x.toFixed(1) + '" y="' + (H - B + 16) + '" text-anchor="middle" font-size="10.5" font-weight="800" fill="#b64858">t' + (idx + 1) + '</text>';
        }
      }
    });

    if (state.mode === "case1" && state.revealed.max) {
      const x4 = xp(4);
      const yv = yp(gapAt(4, "case1", state.x0));
      g += '<line x1="' + x4.toFixed(1) + '" y1="' + yv.toFixed(1) + '" x2="' + x4.toFixed(1) + '" y2="' + y0.toFixed(1) + '" stroke="#e09139" stroke-width="1.3" stroke-dasharray="3 4"></line>';
      g += '<circle cx="' + x4.toFixed(1) + '" cy="' + yv.toFixed(1) + '" r="5" fill="#ffffff" stroke="#e09139" stroke-width="2.2"></circle>';
      g += '<text x="' + (x4 + 6).toFixed(1) + '" y="' + (yv - 7).toFixed(1) + '" font-size="10.5" font-weight="800" fill="#b87522">最大 80 m</text>';
    }

    if (state.mode === "case3" && (state.focus === "min" || state.revealed.cond)) {
      const x4 = xp(4);
      const minV = state.x0 - 24;
      const yv = yp(minV);
      g += '<line x1="' + x4.toFixed(1) + '" y1="' + yv.toFixed(1) + '" x2="' + x4.toFixed(1) + '" y2="' + y0.toFixed(1) + '" stroke="#d78a2e" stroke-width="1.3" stroke-dasharray="3 4" opacity="0.8"></line>';
      g += '<circle cx="' + x4.toFixed(1) + '" cy="' + yv.toFixed(1) + '" r="5" fill="#ffffff" stroke="#d78a2e" stroke-width="2.2"></circle>';
      g += '<text x="' + (x4 + 6).toFixed(1) + '" y="' + (yv - 7).toFixed(1) + '" font-size="10.5" font-weight="800" fill="#a76b1e">D_min = ' + fmt(minV, 1) + ' m</text>';
    }

    const cx = xp(s.t);
    const cy = yp(gapAt(s.t, state.mode, state.x0));
    g += '<line x1="' + cx.toFixed(1) + '" y1="' + T + '" x2="' + cx.toFixed(1) + '" y2="' + (H - B) + '" stroke="#203a5c" stroke-width="1.4" stroke-dasharray="3 4" opacity="0.58"></line>';
    g += '<circle cx="' + cx.toFixed(1) + '" cy="' + cy.toFixed(1) + '" r="4.8" fill="' + (state.mode === "case1" ? "#355d8f" : "#4b6f97") + '" stroke="#ffffff" stroke-width="1.8"></circle>';

    const metricText = state.mode === "case1" ? "S = xA − xB" : "D = xB − xA";
    g += '<text x="' + (L + 5) + '" y="14" font-size="10.5" font-weight="700" fill="#72859d">' + metricText + '</text>';

    dom.gapSvg.innerHTML = g;
    dom.gapFormulaLabel.textContent = metricText;
  }

  function getStatus(s){
    if (state.mode === "case1") {
      const t = s.t;
      if (t < 0.15) return ["初态", "A 在前、B 在后，相距 56.0 m；A 相对 B 更快，间距先增大。"];
      if (t < 3.85) return ["运动中", "vA > vB，间距继续增大；当前间距 " + fmt(s.gap, 1) + " m。"];
      if (t <= 4.15) return ["峰值", "vA = vB = 8 m/s；相对速度为 0，间距达到最大 80 m。"];
      if (t < 7.85) return ["运动中", "vA < vB，间距开始缩小；当前间距 " + fmt(s.gap, 1) + " m。"];
      if (t <= 8.15) return ["关键点", "A 在 t = 8 s 停下；此时 B 还落后 56 m。"];
      if (t < 11.85) return ["追赶", "A 静止，B 继续加速追赶；当前间距 " + fmt(s.gap, 1) + " m。"];
      if (t <= 12.15) return ["相遇", "t = 12 s 时，B 追上 A；两车位置同为 120 m。"];
      return ["已相遇", "B 已在 A 前，间距重新增大；当前间距 " + fmt(s.gap, 1) + " m。"];
    }

    const info = rootInfo(state.x0);
    const eps = 0.16;
    if (state.x0 < 24 - 1e-9) {
      const t1 = info.roots[0];
      const t2 = info.roots[1];
      if (s.t < t1 - eps) return ["追近", "A 在后且更快，间距缩小；当前间距 " + fmt(s.gap, 1) + " m。"];
      if (s.t <= t1 + eps) return ["第一次相遇", "t ≈ " + fmt(t1, 1) + " s，A 追上并超过 B。"];
      if (s.t < t2 - eps) return ["A 已在前", "第一次相遇后，B 逐渐加速追赶；当前间距 " + fmt(s.gap, 1) + " m。"];
      if (s.t <= t2 + eps) return ["第二次相遇", "t ≈ " + fmt(t2, 1) + " s，B 再次追上 A。"];
      return ["远离", "B 在前且更快，间距重新增大。"];
    } else if (state.x0 === 24) {
      if (s.t < 3.84) return ["追近", "A 追近 B；这个间距下最小间距恰好为 0。"];
      if (s.t <= 4.16) return ["临界", "t = 4 s，两车恰好接触一次；没有穿过彼此。"];
      return ["远离", "B 加速后在前，间距增大，不再相遇。"];
    } else {
      if (s.t < 3.84) return ["追近", "A 追近 B，但最小间距仍大于 0。"];
      if (s.t <= 4.16) return ["未追上", "最小间距 = x₀ − 24 > 0，A 追不上 B。"];
      return ["远离", "B 更快且在前，两车间距增大。"];
    }
  }

  function updateReadouts(s){
    dom.rTime.textContent = fmt(s.t, 2) + " s";
    dom.rVrel.textContent = fmt(s.vA - s.vB, 1) + " m/s";
    dom.rVA.textContent = fmt(s.vA, 1) + " m/s";
    dom.rVB.textContent = fmt(s.vB, 1) + " m/s";
    dom.rXA.textContent = fmt(s.xA, 1) + " m";
    dom.rXB.textContent = fmt(s.xB, 1) + " m";
    dom.rGap.textContent = fmt(s.gap, 1) + " m";
    dom.rAhead.textContent = s.gap < 0.16 ? "两车相遇" : (s.xA > s.xB ? "A 在前" : "B 在前");
  }

  function updateTimeUI(s){
    const maxT = state.mode === "case1" ? 14 : 10;
    dom.timeSlider.max = maxT;
    if (Math.abs(parseFloat(dom.timeSlider.value) - s.t) > 0.002) {
      dom.timeSlider.value = s.t.toFixed(2);
    }
    dom.tLabel.textContent = fmt(s.t, 2);
    dom.playBtn.textContent = state.playing ? "暂停" : "播放";
    dom.playStateText.textContent = state.playing ? "播放中" : "暂停";
  }

  function updateStepButtons(){
    document.querySelectorAll(".step-btn").forEach(function(btn){
      btn.classList.toggle("active", btn.dataset.step === state.focus);
    });
    dom.condBtn.classList.toggle("active", state.revealed.cond);
  }

  function updateStepNote(){
    if (state.mode !== "case1") return;
    const map = {
      start: "初态：A 在 56 m 处，B 在 0 m 处；A 更快，间距先增大。",
      max: "vA = vB 时，间距达到峰值。",
      stop: "A 速度减到 0，位置停在 120 m；B 仍以 12 m/s 继续追。",
      meet: "B 追上 A，间距变为 0。"
    };
    dom.stepNote.textContent = map[state.focus] || "拖动时间，观察两车与两张图的同步变化。";
  }

  function updateExploration(){
    if (state.mode !== "case3") return;
    const info = rootInfo(state.x0);
    dom.x0Label.textContent = fmt(state.x0, 1);
    if (Math.abs(parseFloat(dom.x0Slider.value) - state.x0) > 1e-9) {
      dom.x0Slider.value = state.x0;
    }
    dom.rootCount.textContent = String(info.count);
    dom.rootWord.textContent = info.count === 2 ? " 次（可相遇两次）" : (info.count === 1 ? " 次（临界，只接触一次）" : " 次（不相遇）");
    dom.rootCount.classList.remove("count-good", "count-one", "count-zero");
    if (info.count === 2) dom.rootCount.classList.add("count-good");
    else if (info.count === 1) dom.rootCount.classList.add("count-one");
    else dom.rootCount.classList.add("count-zero");
    dom.minNote.innerHTML = 't = 4 s 时，D_min = x₀ − 24 = ' + fmt(state.x0, 1) + ' − 24 = <strong>' + fmt(state.x0 - 24, 1) + ' m</strong>。';
  }

  function updateConclusions(s){
    if (state.mode === "case1") {
      let html = "";
      html += '<div class="result-row">' +
        '<div class="result-label">（1）两车何时相距最大？最大多少？</div>' +
        '<div class="result-value ' + (state.revealed.max ? "" : "pending") + '">' +
          (state.revealed.max ? "t = 4 s，最大间距 80 m" : "点击“找最大间距”后揭示") +
        '</div>' +
        (state.revealed.max ? '<div class="formula-mini">vA = vB → 16 − 2t = 4 + t → t = 4 s<br>S = 56 + 12t − 1.5t² → 80 m</div>' : '') +
      '</div>';

      if (state.revealed.stop) {
        html += '<div class="result-row">' +
          '<div class="result-label">关键转折：A 停下</div>' +
          '<div class="result-value" style="font-size:13px;">t = 8 s，A 停在 120 m；B 在 64 m，还差 56 m。</div>' +
        '</div>';
      }

      html += '<div class="result-row">' +
        '<div class="result-label">（2）经过多长时间两车相遇？</div>' +
        '<div class="result-value ' + (state.revealed.meet ? "" : "pending") + '">' +
          (state.revealed.meet ? "t = 12 s" : "点击“求相遇时间”后揭示") +
        '</div>' +
        (state.revealed.meet ? '<div class="formula-mini">A 停后：120 = 4t + ½t²<br>t² + 8t − 240 = 0 → t = 12 s（另一根为负，舍去）</div>' : '') +
      '</div>';

      dom.conclusions.innerHTML = html;
      return;
    }

    const info = rootInfo(state.x0);
    const countText = info.count === 2 ? "两次相遇" : (info.count === 1 ? "一次接触（临界）" : "不相遇");
    let html = "";
    html += '<div class="result-row">' +
      '<div class="result-label">当前 x₀ = ' + fmt(state.x0, 1) + ' m 时</div>' +
      '<div class="result-value">' + countText + '</div>' +
    '</div>';

    html += '<div class="result-row">' +
      '<div class="result-label">（3）两车可以相遇两次的条件</div>' +
      '<div class="result-value ' + (state.revealed.cond ? "" : "pending") + '">' +
        (state.revealed.cond ? "0 < x₀ < 24 m" : "拖动 x₀ 观察交点，再点“归纳条件”") +
      '</div>' +
      (state.revealed.cond ? '<div class="formula-mini">D = xB − xA = x₀ − 12t + 1.5t²<br>t = 4 s 时，D_min = x₀ − 24<br>要相交两次，需要 D_min < 0，且 x₀ > 0<br>所以：0 < x₀ < 24 m</div>' : '') +
    '</div>';

    dom.conclusions.innerHTML = html;
  }

  function render(){
    const maxT = state.mode === "case1" ? 14 : 10;
    if (state.t > maxT) state.t = maxT;
    const s = motion(state.t, state.mode, state.x0);
    updateTimeUI(s);
    renderScene(s);
    renderVT(s);
    renderGap(s);
    updateReadouts(s);
    updateStatus(s);
    updateStepButtons();
    updateStepNote();
    updateExploration();
    updateConclusions(s);
  }

  function stopAnimation(){
    state.playing = false;
    if (rafId !== null) {
      cancelAnimationFrame(rafId);
      rafId = null;
    }
    lastTs = 0;
  }

  function playLoop(ts){
    if (!state.playing) return;
    if (!lastTs) lastTs = ts;
    const dt = Math.min(0.06, (ts - lastTs) / 1000);
    lastTs = ts;
    const maxT = state.mode === "case1" ? 14 : 10;
    state.t = Math.min(maxT, state.t + dt);
    if (state.t >= maxT - 1e-7) {
      state.t = maxT;
      state.playing = false;
    }
    render();
    if (state.playing) {
      rafId = requestAnimationFrame(playLoop);
    }
  }

  function togglePlay(){
    if (state.playing) {
      stopAnimation();
      render();
      return;
    }
    const maxT = state.mode === "case1" ? 14 : 10;
    if (state.t >= maxT - 1e-6) state.t = 0;
    state.playing = true;
    lastTs = 0;
    rafId = requestAnimationFrame(playLoop);
    render();
  }

  function setMode(mode){
    state.mode = mode;
    state.t = 0;
    state.focus = "none";
    stopAnimation();
    dom.modeCase1.classList.toggle("active", mode === "case1");
    dom.modeCase3.classList.toggle("active", mode === "case3");
    dom.case1Panel.classList.toggle("hidden", mode !== "case1");
    dom.case3Panel.classList.toggle("hidden", mode !== "case3");
    dom.timeSlider.max = mode === "case1" ? 14 : 10;
    dom.timeSlider.value = 0;
    render();
  }

  function bindEvents(){
    dom.modeCase1.addEventListener("click", function(){ setMode("case1"); });
    dom.modeCase3.addEventListener("click", function(){ setMode("case3"); });

    dom.playBtn.addEventListener("click", togglePlay);

    dom.timeSlider.addEventListener("input", function(){
      stopAnimation();
      const maxT = state.mode === "case1" ? 14 : 10;
      state.t = clamp(parseFloat(dom.timeSlider.value), 0, maxT);
      state.focus = "free";
      render();
    });

    document.querySelectorAll("[data-dt]").forEach(function(btn){
      btn.addEventListener("click", function(){
        stopAnimation();
        const maxT = state.mode === "case1" ? 14 : 10;
        state.t = clamp(state.t + parseFloat(btn.dataset.dt), 0, maxT);
        state.focus = "free";
        render();
      });
    });

    dom.resetBtn.addEventListener("click", function(){
      stopAnimation();
      state.t = 0;
      state.focus = "none";
      render();
    });

    document.querySelectorAll(".step-btn").forEach(function(btn){
      btn.addEventListener("click", function(){
        stopAnimation();
        const step = btn.dataset.step;
        state.focus = step;
        if (step === "start") state.t = 0;
        if (step === "max") {
          state.t = 4;
          state.revealed.max = true;
        }
        if (step === "stop") {
          state.t = 8;
          state.revealed.stop = true;
        }
        if (step === "meet") {
          state.t = 12;
          state.revealed.stop = true;
          state.revealed.meet = true;
        }
        if (step === "min") {
          state.t = 4;
        }
        render();
      });
    });

    dom.x0Slider.addEventListener("input", function(){
      stopAnimation();
      state.x0 = parseFloat(dom.x0Slider.value);
      state.focus = "free";
      render();
    });

    document.querySelectorAll("[data-x0]").forEach(function(btn){
      btn.addEventListener("click", function(){
        stopAnimation();
        state.x0 = parseFloat(btn.dataset.x0);
        state.t = 4;
        state.focus = "min";
        render();
      });
    });

    dom.condBtn.addEventListener("click", function(){
      stopAnimation();
      state.revealed.cond = true;
      state.focus = "cond";
      render();
    });
  }

  initTicks();
  bindEvents();
  dom.case1Panel.classList.remove("hidden");
  dom.case3Panel.classList.add("hidden");
  render();
})();
