  function el(tag, attrs){
    const e = document.createElementNS(NS, tag);
    if (attrs) for (const k in attrs) e.setAttribute(k, attrs[k]);
    return e;
  }
  function setLine(e, x1, y1, x2, y2){
    e.setAttribute("x1", X(x1));
    e.setAttribute("y1", Y(y1));
    e.setAttribute("x2", X(x2));
    e.setAttribute("y2", Y(y2));
  }
  function setPt(e, x, y){
    e.setAttribute("cx", X(x));
    e.setAttribute("cy", Y(y));
  }
  function setLbl(e, x, y, dx, dy, txt){
    e.setAttribute("x", X(x) + dx);
    e.setAttribute("y", Y(y) + dy);
    if (txt !== undefined) e.textContent = txt;
  }
  function polyString(pts, fx, fy){
    return pts.map(p => `${fx(p[0]).toFixed(2)},${fy(p[1]).toFixed(2)}`).join(" ");
  }
  function setPoly(e, pts){
    e.setAttribute("points", polyString(pts, X, Y));
  }
  function setPolyWith(e, pts, fx, fy){
    e.setAttribute("points", polyString(pts, fx, fy));
  }

  function derive(s){
    const p = s.p;
    const yP = p*p/8;
    const distPQ = Math.hypot(p, yP - 2);

    const r = s.r;
    const a = 8 / (r + 1);
    const b = -8;
    const c = 8 * r / (r + 1);
    const yA = a*a/8;
    const yB = 8;
    const yC = c*c/8;
    const yG = (yA + yB + yC) / 3;
    const yM = a;
    const yN = c;

    const S2 = 0.5 * a * (yG - yM);
    const S3 = 0.5 * c * (yN - yG);
    const det = (b - a) * (yC - yA) - (c - a) * (yB - yA);
    const S1 = 0.5 * Math.abs(det);
    const R = (S2 + S3) / S1;

    return {p, yP, distPQ, r, a, b, c, yA, yB, yC, yG, yM, yN, S1, S2, S3, R};
  }

    els.S2lbl.setAttribute("x", X(d.a * 0.55));
    els.S2lbl.setAttribute("y", Y((d.yA + d.yM + d.yG) / 3) + 4);
    els.S2lbl.textContent = "S2";

    els.S3lbl.setAttribute("x", X(d.c * 0.55));
    els.S3lbl.setAttribute("y", Y((d.yC + d.yN + d.yG) / 3) + 4);
    els.S3lbl.textContent = "S3";

    applyVisibility(s.step);
    updateRangeGraph(d);
    updateNorm(d);
  }

  function applyVisibility(step){
    const is1 = step === 1;
    els.s1.style.display = is1 ? "" : "none";
    els.directrixLayer.style.display = is1 ? "" : "none";
    els.vertexLbl.style.display = is1 ? "" : "none";
    els.tri.style.display = is1 ? "none" : "";

    const showMN = step >= 3;
    [els.Mdot, els.Ndot, els.Mlbl, els.Nlbl].forEach(e => {
      e.style.display = showMN ? "" : "none";
    });

    const s3 = step === 3;
    [els.segAM, els.segMB, els.simA, els.simB, els.projA3, els.projB3,
     els.a3Lbl, els.s3Lbl, els.A1lbl, els.B1lbl].forEach(e => {
      e.style.display = s3 ? "" : "none";
    });

    const showArea = step >= 4;
    [els.polyS2, els.polyS3, els.S1lbl, els.S2lbl, els.S3lbl,
     els.projA4, els.projC4, els.a4Lbl, els.c4Lbl,

     els.acBaseHi, els.alt4, els.acLbl, els.altLbl].forEach(e => {
      e.style.display = showArea ? "" : "none";
    });

    els.polyS1.setAttribute("class", showArea ? "area-s1" : "area-outline");
  }

  function updateNorm(d){
    if (!els.norm) return;
    const s0 = 12;
    const lam = 8 / s0;
    const r = d.r;
    const a0 = s0 / (r + 1);
    const b0 = -s0;
    const c0 = s0 * r / (r + 1);
    const yA0 = a0*a0/8;
    const yB0 = 18;
    const yC0 = c0*c0/8;
    const yG0 = (yA0 + yB0 + yC0) / 3;

    const a = lam * a0;
    const b = -8;
    const c = lam * c0;
    const yA = d.yA;
    const yB = d.yB;
    const yC = d.yC;
    const yG = d.yG;

    const LX = 70, RX = 270, baseY = 160, xScale = 4.5, yScale = 4.0;
    const lx = x => LX + xScale * x;
    const ly = y => baseY - yScale * y;
    const rx = x => RX + xScale * x;
    const ry = y => baseY - yScale * y;

    setPolyWith(els.norm.lTri, [[a0, yA0], [b0, yB0], [c0, yC0]], lx, ly);
    setPolyWith(els.norm.rTri, [[a, yA], [b, yB], [c, yC]], rx, ry);

    function placePt(e, x, y){
      e.setAttribute("cx", x);
      e.setAttribute("cy", y);
    }
    function placeLbl(e, x, y, anchor){
      e.setAttribute("x", x);
      e.setAttribute("y", y);
      if (anchor) e.setAttribute("text-anchor", anchor);
    }

    placePt(els.norm.lA, lx(a0), ly(yA0));
    placePt(els.norm.lB, lx(b0), ly(yB0));
    placePt(els.norm.lC, lx(c0), ly(yC0));
    placePt(els.norm.lG, lx(0), ly(yG0));
    placePt(els.norm.rA, rx(a), ry(yA));
    placePt(els.norm.rB, rx(b), ry(yB));
    placePt(els.norm.rC, rx(c), ry(yC));
    placePt(els.norm.rG, rx(0), ry(yG));

    placeLbl(els.norm.lAl, lx(a0) + 4, ly(yA0) - 4, "start");
    placeLbl(els.norm.lBl, lx(b0) + 4, ly(yB0) - 4, "start");
    placeLbl(els.norm.lCl, lx(c0) + 4, ly(yC0) - 4, "start");
    placeLbl(els.norm.lGl, LX + 6, ly(yG0) - 3, "start");

    placeLbl(els.norm.rAl, rx(a) + 4, ry(yA) - 4, "start");
    placeLbl(els.norm.rBl, rx(b) + 4, ry(yB) - 4, "start");
    placeLbl(els.norm.rCl, rx(c) + 4, ry(yC) - 4, "start");
    placeLbl(els.norm.rGl, RX + 6, ry(yG) - 3, "start");
  }

  function updateRangeGraph(d){
    if (!els.range) return;
    const cx = gx(d.r);
    const cy = gy(d.R);
    const near = d.r < 2.7;
    const dx = near ? 12 : 0;
    const dy = near ? 14 : 0;
    const px = cx + dx;
    const py = cy + dy;

    els.range.dot.setAttribute("cx", px);
    els.range.dot.setAttribute("cy", py);

    els.range.leader.style.display = near ? "" : "none";
    els.range.leader.setAttribute("x1", cx);
    els.range.leader.setAttribute("y1", cy);
    els.range.leader.setAttribute("x2", px);
    els.range.leader.setAttribute("y2", py);

    els.range.actual.style.display = near ? "" : "none";
    els.range.actual.setAttribute("cx", cx);
    els.range.actual.setAttribute("cy", cy);

    let labelX = px + 8;
    let anchor = "start";
    if (px > 285){
      labelX = px - 8;
      anchor = "end";
    }
    els.range.valLbl.setAttribute("x", labelX);
    els.range.valLbl.setAttribute("y", py - 7);
    els.range.valLbl.setAttribute("text-anchor", anchor);
    els.range.valLbl.textContent = "R=" + d.R.toFixed(4);
  }

  function clampP(v, prev){
    v = Math.max(-7, Math.min(7, v));
    if (Math.abs(v) < 0.6){
      if (v > 0) v = 0.6;
      else if (v < 0) v = -0.6;
      else v = prev < 0 ? -0.6 : 0.6;
    }
    return v;
  }

  function setState(partial){
    const prevP = state.p;
    Object.assign(state, partial);
    state.step = Math.max(1, Math.min(5, state.step));
    state.p = clampP(state.p, prevP);
    state.r = Math.max(2.25, Math.min(10, state.r));
    if (state.step === 2) state.r = 3;

    const d = derive(state);
    render(state, d);
    updateSidebar(state, d);
    updateControls(state, d);
  }

  function bindEvents(){
    document.getElementById("pSlider").addEventListener("input", function(){
      setState({p: parseFloat(this.value)});
    });
    document.getElementById("rSlider").addEventListener("input", function(){
      setState({r: parseFloat(this.value)});
    });
    document.getElementById("prevBtn").addEventListener("click", () => setState({step: state.step - 1}));
    document.getElementById("nextBtn").addEventListener("click", () => setState({step: state.step + 1}));
    document.getElementById("nearBtn").addEventListener("click", () => setState({r: 2.25}));
    document.getElementById("resetBtn").addEventListener("click", () => setState({step: 1, p: 4.5, r: 3}));
    document.querySelectorAll(".tab").forEach(t => {
      t.addEventListener("click", () => setState({step: +t.dataset.step}));
    });
  }

  function init(){
    buildMain();
    buildNorm();
    buildRange();
    bindEvents();
    setState({step: 1, p: 4.5, r: 3});
  }

  document.addEventListener("DOMContentLoaded", init);
})();
