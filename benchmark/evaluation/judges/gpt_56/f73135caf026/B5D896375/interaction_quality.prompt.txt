      roundRect(ctx, x0 - 14, rwTop, (x1 + 14) - (x0 - 14), rwBot - rwTop, 9);
      ctx.clip();
      ctx.fillStyle = 'rgba(245,158,11,0.26)';
      ctx.fillRect(X(SLAST), rwTop, X(STOT) - X(SLAST), rwBot - rwTop);
      ctx.restore();

      ctx.strokeStyle = '#f59e0b';
      ctx.lineWidth = 2;
      ctx.setLineDash([6,5]);
      ctx.beginPath();
      ctx.moveTo(X(SLAST), rwTop - 12);
      ctx.lineTo(X(SLAST), rwBot + 6);
      ctx.stroke();
      ctx.setLineDash([]);

      ctx.fillStyle = '#b45309';
      ctx.font = '700 12px ' + FONT;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'bottom';
      ctx.fillText('t = 2 s 从这里算起', X(SLAST), rwTop - 16);
    }

    /* --- 终点虚线 --- */
    if(state.showStop){
      ctx.strokeStyle = 'rgba(37,99,235,0.55)';
      ctx.lineWidth = 2;
      ctx.setLineDash([5,5]);
      ctx.beginPath();
      ctx.moveTo(X(STOT), rwTop - 12);
      ctx.lineTo(X(STOT), rwBot + 6);
      ctx.stroke();
      ctx.setLineDash([]);
      ctx.fillStyle = '#1d4ed8';
      ctx.font = '700 12px ' + FONT;
      ctx.textAlign = 'right';
      ctx.textBaseline = 'bottom';
      ctx.fillText('静止处', X(STOT) + 6, rwTop - 16);
    }

    /* --- 跑道中线 --- */
    ctx.save();
    roundRect(ctx, x0 - 14, rwTop, (x1 + 14) - (x0 - 14), rwBot - rwTop, 9);
    ctx.clip();
    ctx.strokeStyle = 'rgba(255,255,255,0.85)';
    ctx.lineWidth = 2;
    ctx.setLineDash([15,13]);
    ctx.beginPath();
    ctx.moveTo(x0 - 20, rwMid + 24);
    ctx.lineTo(x1 + 20, rwMid + 24);
    ctx.stroke();
    ctx.setLineDash([]);
    ctx.restore();

    /* --- 刻度 --- */
    ctx.font = '11px ' + FONT;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    [0, 36, 72, 108].forEach(function(m){
      if(m === 108 && !state.showArea) return;
      const x = X(m);
      ctx.strokeStyle = '#9aa9bd';
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(x, rwBot);
      ctx.lineTo(x, rwBot + 7);
      ctx.stroke();
      ctx.fillStyle = '#7c8ba1';
      ctx.fillText(m + ' m', x, rwBot + 11);
    });

    /* --- 已滑行距离条 --- */
    ctx.fillStyle = 'rgba(148,163,184,0.45)';
    ctx.fillRect(x0, rwBot + 30, x1 - x0, 5);
    ctx.fillStyle = '#2563eb';
    ctx.fillRect(x0, rwBot + 30, Math.max(0, X(sNow) - x0), 5);

    /* --- 速度矢量（绿） --- */
    if(vNow > 0.05){
      const ay = rwMid - 27;
      const axA = clamp(X(sNow) + 8, x0 + 4, x1 - 52);
      const len = Math.max(16, vNow * 1.55);
      const axB = Math.min(axA + len, x1 - 4);
      arrow(ctx, axA, ay, axB, ay, '#2563eb', 3);
      ctx.fillStyle = '#1d4ed8';
      ctx.font = '700 12px ' + FONT;
      ctx.textAlign = 'left';
      ctx.textBaseline = 'bottom';
      const vLabelX = Math.min(axA - 2, x1 - 74);
      ctx.fillText('v = ' + vNow.toFixed(1) + ' m/s', vLabelX, ay - 8);
    }

    /* --- 加速度矢量（红，与 v 反向；v=0 时运动结束，不再画 a） --- */
    if(vNow > 0.05){
      const ayA = rwMid + 30;
      const aTailX = clamp(X(sNow) + 6, x0 + 48, x1 - 8);
      const aHeadX = aTailX - 44;
      arrow(ctx, aTailX, ayA, aHeadX, ayA, '#ef4444', 3);
      ctx.fillStyle = '#dc2626';
      ctx.font = '700 12px ' + FONT;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'bottom';
      const aLabelX = clamp((aHeadX + aTailX) / 2, x0 + 72, x1 - 72);
      ctx.fillText('a = 6 m/s²，方向向左', aLabelX, ayA - 8);
    }

    /* --- 飞机 --- */
    drawPlane(ctx, X(sNow), rwMid - 2, 0.8);

    /* --- 左下角时间 --- */
    ctx.fillStyle = '#94a3b8';
    ctx.font = '11.5px ' + FONT;
    ctx.textAlign = 'right';
    ctx.textBaseline = 'top';
    ctx.fillText('t = ' + t.toFixed(1) + ' s', w - 12, 12);
  }

  /* ================= v–t 图像 ================= */
  function drawChart(w, h){
    const ctx = cctx;
    ctx.clearRect(0, 0, w, h);

    const tMax = 7.1, vMax = 42;
    const L = 52, R = w - 20, Tp = 18, Bt = h - 42;
    const X = t => L + (t/tMax)*(R-L);
    const Y = v => Bt - (v/vMax)*(Bt-Tp);

    /* 网格：揭示停止时间前，只画到当前时刻 */
    ctx.strokeStyle = '#e9eff8';
    ctx.lineWidth = 1;
    const tNow = state.t;
    const tShown = state.showStop ? 6 : Math.min(6, tNow);
    for(let v = 0; v <= 36; v += 12){
      ctx.beginPath(); ctx.moveTo(L, Y(v)); ctx.lineTo(R, Y(v)); ctx.stroke();
    }
    for(let t = 0; t <= tShown + 0.001; t += 2){
      ctx.beginPath(); ctx.moveTo(X(t), Tp); ctx.lineTo(X(t), Bt); ctx.stroke();
    }

    /* 当前累计位移：只填充 0→t 的面积 */
    const vNow = vOf(tNow);
    if(tNow > 0.001){
      ctx.beginPath();
      ctx.moveTo(X(0), Y(0));
      ctx.lineTo(X(0), Y(36));
      ctx.lineTo(X(tNow), Y(vNow));
      ctx.lineTo(X(tNow), Y(0));
      ctx.closePath();
      ctx.fillStyle = 'rgba(37,99,235,0.18)';
      ctx.fill();
    }

    /* 完整三角形的轮廓（表示整段 0–6 s 的总面积） */
    if(state.showArea){
      ctx.beginPath();
      ctx.moveTo(X(0), Y(0));
      ctx.lineTo(X(0), Y(36));
      ctx.lineTo(X(6), Y(0));
      ctx.closePath();
      ctx.strokeStyle = 'rgba(37,99,235,0.55)';
      ctx.lineWidth = 1.8;
      ctx.setLineDash([6,5]);
      ctx.stroke();
      ctx.setLineDash([]);
    }

    /* 最后 4 s 的面积 */
    if(state.showLast4){
      ctx.beginPath();
      ctx.moveTo(X(2), Y(0));
      ctx.lineTo(X(2), Y(24));
      ctx.lineTo(X(6), Y(0));
      ctx.closePath();
      ctx.fillStyle = 'rgba(245,158,11,0.45)';
      ctx.fill();
      ctx.strokeStyle = '#f59e0b';
      ctx.lineWidth = 2;
      ctx.stroke();
    }

    /* 坐标轴 */
    ctx.strokeStyle = '#94a3b8';
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    ctx.moveTo(L, Tp - 6);
    ctx.lineTo(L, Bt);
    ctx.lineTo(R, Bt);
    ctx.stroke();

    /* 轴末端小箭头 */
    ctx.fillStyle = '#94a3b8';
    ctx.beginPath();
    ctx.moveTo(L, Tp - 10); ctx.lineTo(L - 4, Tp - 2); ctx.lineTo(L + 4, Tp - 2);
    ctx.closePath(); ctx.fill();
    ctx.beginPath();
    ctx.moveTo(R + 5, Bt); ctx.lineTo(R - 3, Bt - 4); ctx.lineTo(R - 3, Bt + 4);
    ctx.closePath(); ctx.fill();

    /* 刻度文字 */
    ctx.fillStyle = '#8b9ab0';
    ctx.font = '11px ' + FONT;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    for(let t = 0; t <= (state.showStop ? 6 : Math.min(6, state.t)) + 0.001; t += 2){
      ctx.fillText(t + '', X(t), Bt + 6);
    }
    ctx.textAlign = 'right';
    ctx.textBaseline = 'middle';
    for(let v = 0; v <= 36; v += 12){
      ctx.fillText(v + '', L - 8, Y(v));
    }

    /* 轴名 */
    ctx.fillStyle = '#64748b';
    ctx.font = '12px ' + FONT;
    ctx.textAlign = 'left';
    ctx.textBaseline = 'top';
    ctx.fillText('t / s', R - 34, Bt + 18);
    ctx.textAlign = 'left';
    ctx.textBaseline = 'bottom';
    ctx.fillText('v /(m·s⁻¹)', L + 6, Tp + 2);

    /* 图线：未揭示停止时间前，只画到当前时刻 */
    const tLine = state.showStop ? 6 : Math.min(6, state.t);
    ctx.beginPath();
    ctx.moveTo(X(0), Y(36));
    if(tLine > 0.001) ctx.lineTo(X(tLine), Y(vOf(tLine)));
    ctx.strokeStyle = '#2563eb';
    ctx.lineWidth = 2.8;
    ctx.lineCap = 'round';
    if(tLine > 0.001) ctx.stroke();

    /* 端点 */
    function dot(x, y, color, r){
      ctx.beginPath();
      ctx.arc(x, y, r, 0, Math.PI*2);
      ctx.fillStyle = color;
      ctx.fill();
      ctx.strokeStyle = '#fff';
      ctx.lineWidth = 2;
      ctx.stroke();
    }
    dot(X(0), Y(36), '#2563eb', 4.5);
    if(state.showStop) dot(X(6), Y(0), '#2563eb', 4.5);

    if(state.showLast4){
      dot(X(2), Y(24), '#f59e0b', 6);
      ctx.fillStyle = '#b45309';
      ctx.font = '700 11.5px ' + FONT;
      ctx.textAlign = 'left';
      ctx.textBaseline = 'middle';
      ctx.fillText('v = 24', X(2) + 11, Y(24) - 12);
    }

    /* 当前时刻的标记：空心环，避免遮住橙色 (2 s, 24 m/s) 标记 */
    const t = state.t, v = vOf(t);
    ctx.setLineDash([4,4]);
    ctx.strokeStyle = '#b6c2d4';
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.moveTo(X(t), Y(v));
    ctx.lineTo(X(t), Bt);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(L, Y(v));
    ctx.lineTo(X(t), Y(v));
    ctx.stroke();
    ctx.setLineDash([]);

    ctx.beginPath();
    ctx.arc(X(t), Y(v), 8.6, 0, Math.PI*2);
    ctx.strokeStyle = '#1d4ed8';
    ctx.lineWidth = 2.2;
    ctx.stroke();

    /* 面积文字 */
    if(state.showArea){
      ctx.fillStyle = '#1d4ed8';
      ctx.font = '700 12.5px ' + FONT;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText('整段总面积 108 m', X(3.2), Y(27));
    }
    if(state.showLast4){
      ctx.fillStyle = '#b45309';
      ctx.font = '700 12.5px ' + FONT;
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText('2–6 s 位移 48 m', X(4.1), Y(9));
    }
    /* 倒放视图已移到独立卡片 */
  }

  function drawReverse(w, h){
    const ctx = rctx;
    ctx.clearRect(0, 0, w, h);

    const L = 54, R = w - 26, Tp = 16, Bt = h - 30;
    const X = t => L + (t / 4.4) * (R - L);
    const Y = v => Bt - (v / 28) * (Bt - Tp);

    ctx.strokeStyle = '#f0e6d6';
    ctx.lineWidth = 1;
    for(let v = 0; v <= 24; v += 12){
      ctx.beginPath(); ctx.moveTo(L, Y(v)); ctx.lineTo(R, Y(v)); ctx.stroke();
    }
    for(let t = 0; t <= 4; t += 2){
      ctx.beginPath(); ctx.moveTo(X(t), Tp); ctx.lineTo(X(t), Bt); ctx.stroke();
    }

    ctx.beginPath();
    ctx.moveTo(X(0), Y(0));
    ctx.lineTo(X(4), Y(0));
    ctx.lineTo(X(4), Y(24));
    ctx.closePath();
    ctx.fillStyle = 'rgba(245,158,11,0.38)';
    ctx.fill();
    ctx.strokeStyle = '#b45309';
    ctx.lineWidth = 2.2;
    ctx.stroke();

    ctx.strokeStyle = '#94a3b8';
    ctx.lineWidth = 1.4;

    ctx.lineTo(L, Bt);

    ctx.fillStyle = '#8b9ab0';
    ctx.font = '12px ' + FONT;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    ctx.fillText('0', X(0), Bt + 6);

    ctx.fillText('4 s', X(4), Bt + 6);

    ctx.textBaseline = 'middle';

    ctx.fillText('24', L - 7, Y(24));

    ctx.textBaseline = 'bottom';
    ctx.fillText('v /(m·s⁻¹)', L + 6, Tp + 2);
    ctx.textAlign = 'right';
    ctx.textBaseline = 'top';
    ctx.fillText('t / s', R, Bt + 20);
    ctx.fillStyle = '#b45309';
    ctx.font = '700 12.5px ' + FONT;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('面积 = 48 m', X(2.35), Y(12));
  }

  /* ================= 总渲染 ================= */
  function render(){
    const a = sizeCanvas(sceneCv, sctx);
    drawScene(a.w, a.h);
    const b = sizeCanvas(chartCv, cctx);
    drawChart(b.w, b.h);
    document.getElementById('reverseCard').style.display = state.showReverse ? 'block' : 'none';
    const c = sizeCanvas(reverseCv, rctx);
    drawReverse(c.w, c.h);
    updateReadout();
  }

  function updateReadout(){
    const t = state.t;
    const v = vOf(t);
    const s = sOf(t);
    document.getElementById('rT').textContent = t.toFixed(1) + ' s';
    document.getElementById('rV').textContent = v.toFixed(1) + ' m/s';
    document.getElementById('rS').textContent = s.toFixed(1) + ' m';
    document.getElementById('rV2').textContent = v.toFixed(1) + ' m/s';
    document.getElementById('rS2').textContent = s.toFixed(1) + ' m';
    timeLabel.textContent = 't = ' + t.toFixed(1) + ' s';
  }

  /* ================= 时间控制 ================= */
  let tweenRAF = null;

  function setTime(t){
    state.t = clamp(t, 0, TSTOP);
    slider.value = String(state.t);
  }

  function animateTimeTo(target, dur){
    if(tweenRAF) cancelAnimationFrame(tweenRAF);
    const from = state.t;
    const t0 = performance.now();
    dur = dur || 420;
    function frame(now){
      const p = Math.min(1, (now - t0) / dur);
      const e = 1 - Math.pow(1 - p, 3);
      setTime(from + (target - from) * e);
      render();
      if(p < 1) tweenRAF = requestAnimationFrame(frame);
      else tweenRAF = null;
    }
    tweenRAF = requestAnimationFrame(frame);
  }

  let playRAF = null;
  let lastNow = 0;
  const SPEED = 1.5; // 模拟秒 / 真实秒

  function stopPlay(){
    state.playing = false;
    if(playRAF) cancelAnimationFrame(playRAF);
    playRAF = null;
    playBtn.textContent = state.t >= TSTOP - 0.01 ? '重播' : '播放';
  }

  function startPlay(){
    if(tweenRAF){ cancelAnimationFrame(tweenRAF); tweenRAF = null; }
    state.playing = true;
    playBtn.textContent = '暂停';
    lastNow = performance.now();
    function loop(now){
      if(!state.playing) return;
      const dt = (now - lastNow) / 1000;
      lastNow = now;
      let t = state.t + dt * SPEED;
      if(t >= TSTOP){ t = TSTOP; }
      setTime(t);
      render();
      if(t >= TSTOP - 0.0001){ stopPlay(); return; }
      playRAF = requestAnimationFrame(loop);
    }
    playRAF = requestAnimationFrame(loop);
  }

  /* ================= 步骤控制 ================= */
  function buildStepButtons(){
    STEPS.forEach(function(st, i){
      const b = document.createElement('button');
      b.className = 'step-btn' + (i === 0 ? ' active' : '');
      b.textContent = st.btn;
      b.addEventListener('click', function(){ setStep(i); });
      stepButtons.appendChild(b);
    });
  }

  function setStep(i){
    state.step = i;
    state.showStop = i >= 1;
    state.showArea = i >= 2;
    state.showLast4 = i >= 3;
    state.showReverse = i >= 4;

    const btns = stepButtons.querySelectorAll('.step-btn');
    btns.forEach(function(b, k){ b.classList.toggle('active', k === i); });

    stepText.textContent = STEPS[i].text;
    stepFormula.textContent = STEPS[i].formula;
    stepFormula.classList.toggle('warm', i >= 3);

    if(state.playing) stopPlay();

    const targetT = [0, 0, 6, 2, 2][i];
    animateTimeTo(targetT, 480);
  }

  /* ================= 事件绑定 ================= */
  slider.addEventListener('input', function(){
    if(state.playing) stopPlay();
    if(tweenRAF){ cancelAnimationFrame(tweenRAF); tweenRAF = null; }
    setTime(parseFloat(slider.value));
    render();
  });

  playBtn.addEventListener('click', function(){
    if(state.playing){
      stopPlay();
    } else {
      if(state.t >= TSTOP - 0.01) setTime(0);
      startPlay();
    }
  });

  resetBtn.addEventListener('click', function(){
    if(state.playing) stopPlay();
    if(tweenRAF){ cancelAnimationFrame(tweenRAF); tweenRAF = null; }
    state.showStop = false;
    state.showArea = false;
    state.showLast4 = false;
    state.showReverse = false;
    state.step = 0;
    const btns = stepButtons.querySelectorAll('.step-btn');
    btns.forEach(function(b, k){ b.classList.toggle('active', k === 0); });
    stepText.textContent = STEPS[0].text;
    stepFormula.textContent = STEPS[0].formula;
    stepFormula.classList.remove('warm');
    playBtn.textContent = '播放';
    setTime(0);
    render();
  });

  let resizeTimer = null;
  window.addEventListener('resize', function(){
    if(resizeTimer) clearTimeout(resizeTimer);
    resizeTimer = setTimeout(render, 110);
  });

  /* ================= 启动 ================= */
  buildStepButtons();
  stepText.textContent = STEPS[0].text;
  stepFormula.textContent = STEPS[0].formula;
  setTime(0);
  render();

  // 首帧字体/布局稳定后再渲染一次
  setTimeout(render, 60);
  window.addEventListener('load', render);

})();
