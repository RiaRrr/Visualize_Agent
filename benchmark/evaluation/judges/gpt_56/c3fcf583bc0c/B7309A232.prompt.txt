    for (let y = -30; y <= 0; y += 5) {

    axes += '<line x1="' + x0 + '" y1="' + plot.bottom + '" x2="' + x0 + '" y2="' + plot.top + '" stroke="#8a98aa" stroke-width="1.5" marker-end="url(#axisArrow)"/>';

    for (let y = -30; y <= -5; y += 5) {
      const yy = sy(y);

      axes += '<text x="' + (x0 - 9) + '" y="' + (yy + 4) + '" text-anchor="end" class="tick-label">' + String(y).replace("-", "−") + '</text>';
    }

    axes += '<text x="' + (plot.right + 18) + '" y="' + (y0 + 5) + '" class="axis-name" text-anchor="start">x</text>';
    axes += '<text x="' + (x0 + 10) + '" y="' + (plot.top + 13) + '" class="axis-name">y</text>';
    axesLayer.innerHTML = axes;

    let pathData = "";

      const px = sx(x);

      pathData += (i === 0 ? "M" : "L") + px.toFixed(2) + "," + py.toFixed(2) + " ";

      '<path d="M ' + (rightEndX - 5) + ' ' + (rightEndY - 2) + ' L ' + (rightEndX + 14) + ' ' + (rightEndY + 11) + '" fill="none" stroke="#2563eb" stroke-width="3.4" stroke-linecap="round" marker-end="url(#extendArrow)"/>' +
      '<text x="' + sx(1) + '" y="' + (plot.bottom - 6) + '" text-anchor="middle" class="svg-note">曲线继续向下延伸</text>';

    rangeBand.setAttribute("x", plot.left);
    rangeBand.setAttribute("y", sy(-13));
    rangeBand.setAttribute("width", plot.right - plot.left);
    rangeBand.setAttribute("height", sy(-30) - sy(-13));

    axisSym.setAttribute("x1", sx(1));

  function updateCard(x, y, d, d2) {
    let title = "";
    let body = "";

    if (state.step === 0) {
      title = "① 观察：点 P 怎样改变 y";
      body =
        '<p>拖动滑块，让图象上的点 <b>P</b> 移动。P 的纵坐标就是函数值 <b>y</b>。</p>' +
        '<div class="live-box">x = <b>' + fmt(x) + '</b> &nbsp;→&nbsp; y = <b>' + fmt(y) + '</b></div>' +
        '<p class="hint">先观察 y 有没有可能比图上的最高点更大。</p>';
    } else if (state.step === 1) {
      title = "② 配方：找到最高点";
      body =
        '<p>把函数改写成顶点式，就能直接读出顶点。</p>' +
        '<div class="derivation">' +
          'y = −x<sup>2</sup> + 2x − 14<br>' +
          '= −(x<sup>2</sup> − 2x) − 14<br>' +
          '= −[(x − 1)<sup>2</sup> − 1] − 14<br>' +
          '= −(x − 1)<sup>2</sup> − 13' +
        '</div>' +
        '<p>图上顶点是 <b>(1, −13)</b>，对称轴是 <b>x = 1</b>。</p>';
    } else if (state.step === 2) {
      title = "③ 把平方看成“离 1 的距离”";
      body =
        '<p>令 d = |x − 1|。顶点水平线上的橙色线段表示 d，紫色竖直下降量表示 d<sup>2</sup>。</p>' +
        '<div class="derivation">y = −(x − 1)<sup>2</sup> − 13 = −13 − d<sup>2</sup></div>' +
        '<div class="live-box">d = <b>' + fmt(d) + '</b> &nbsp; d<sup>2</sup> = <b>' + fmt(d2) + '</b> &nbsp; y = <b>' + fmt(y) + '</b></div>' +
        '<p class="hint">平方 d<sup>2</sup> 永远 ≥ 0，所以 y 只能等于 −13 再减去一个非负数。</p>';
    } else {
      title = "④ 得出值域";
      body =
        '<p>无论 x 取什么值，都有 y = −13 − d<sup>2</sup> ≤ −13。并且当 x 取遍全体实数时，d = |x − 1| 取遍 [0, ∞)，所以 y 取遍 (−∞, −13]。</p>' +
        '<div class="derivation">对任意 t ≤ −13，令 d = √(−13 − t)，取 x = 1 ± d，则 y = −13 − d<sup>2</sup> = t。</div>' +
        '<div class="live-box">当前 y = <b>' + fmt(y) + '</b>；当 x = 1 时，y = <b>−13</b>。</div>' +
        '<div class="final-box">值域：y ∈ (−∞, −13]</div>' +
        '<p class="hint">−13 可取到；−∞ 只是向下无限延伸，不是能取到的数。</p>';
    }

    stepTitle.textContent = title;
    stepBody.innerHTML = body;
  }

  function render() {
    const x = state.x;
    const y = f(x);
    const d = Math.abs(x - 1);
    const d2 = d * d;
    const xPix = sx(x);
    const yPix = sy(y);

    xSlider.value = x;
    xReadout.textContent = "x = " + fmt(x);
    yReadout.textContent = "y = " + fmt(y);

    pointDot.setAttribute("cx", xPix);
    pointDot.setAttribute("cy", yPix);

    verticalGuide.setAttribute("x1", xPix);
    verticalGuide.setAttribute("x2", xPix);
    verticalGuide.setAttribute("y1", yPix);
    verticalGuide.setAttribute("y2", sy(0));

    xTickDot.setAttribute("cx", xPix);

    const labelRight = x <= 3.6;
    pointLabel.setAttribute("x", xPix + (labelRight ? 12 : -12));
    pointLabel.setAttribute("y", yPix - 12);
    pointLabel.setAttribute("text-anchor", labelRight ? "start" : "end");
    pointLabel.textContent = "P (" + fmt(x) + ", " + fmt(y) + ")";

    baseLine.setAttribute("x2", xPix);
    dropLine.setAttribute("x1", xPix);
    dropLine.setAttribute("x2", xPix);
    dropLine.setAttribute("y1", sy(-13));
    dropLine.setAttribute("y2", yPix);

    const midX = (sx(1) + xPix) / 2;
    const dSmall = d < 0.8;
    dLabel.setAttribute("x", dSmall ? (x >= 1 ? xPix + 12 : xPix - 12) : midX);
    dLabel.setAttribute("text-anchor", dSmall ? (x >= 1 ? "start" : "end") : "middle");
    dLabel.setAttribute("y", sy(-13) + (dSmall ? 32 : -10));
    dLabel.textContent = "d = " + fmt(d);

    dropLabel.setAttribute("x", xPix + (x >= 1 ? -12 : 12));
    dropLabel.setAttribute("y", (sy(-13) + yPix) / 2 + 5);
    dropLabel.setAttribute("text-anchor", x >= 1 ? "end" : "start");
    dropLabel.textContent = "d² = " + fmt(d2);



    show(axisSym, state.step >= 1);
    show(axisSymLabel, state.step >= 1);
    show(vertexLevel, state.step >= 1);
    show(vertexLevelLabel, state.step >= 1);
    show(vertexDot, state.step >= 1);
    show(vertexLabel, state.step >= 1);

    show(baseLine, state.step >= 2);
    show(dropLine, state.step >= 2);
    show(dropLabel, state.step >= 2);

    show(dLabel, state.step >= 2);

    show(rangeBand, state.step >= 3);

    rangePanel.style.display = state.step >= 3 ? "block" : "none";
    rangeCurrentDot.setAttribute("cx", rangeX(y));
    rangeCurrentDot.style.opacity = state.step >= 3 ? "1" : "0";

    updateCard(x, y, d, d2);

    if (state.step === 0) {
      answerChip.textContent = "值域：待探索";
      answerChip.classList.remove("revealed");
    } else if (state.step === 1) {
      answerChip.textContent = "顶点：最高点 (1, −13)";
      answerChip.classList.remove("revealed");
    } else if (state.step === 2) {
      answerChip.textContent = "关键：y = −13 − d²";
      answerChip.classList.remove("revealed");
    } else {
      answerChip.textContent = "值域：(−∞, −13]";
      answerChip.classList.add("revealed");
    }

    document.querySelectorAll(".step-btn").forEach(function (btn) {
      btn.classList.toggle("active", Number(btn.dataset.step) === state.step);
    });
  }

  function setX(value, fromPlay) {
    state.x = Math.max(-2.5, Math.min(4.5, value));
    if (fromPlay) {
      xSlider.value = state.x;
    }
    render();
  }

  let playRAF = null;
  let playStart = 0;
  let playPhase = 0;

  function stopPlay() {
    if (state.playing) {
      state.playing = false;
      if (playRAF) {
        cancelAnimationFrame(playRAF);
        playRAF = null;
      }
    }
    playBtn.textContent = "自动扫描";
  }

  function animatePlay(now) {
    if (!state.playing) return;
    const elapsed = (now - playStart) / 1000;
    const x = 1 + 3.5 * Math.sin(playPhase + elapsed * 1.15);
    setX(x, true);
    playRAF = requestAnimationFrame(animatePlay);
  }

  function startPlay() {
    state.playing = true;
    playBtn.textContent = "暂停扫描";
    playStart = performance.now();
    const ratio = Math.max(-1, Math.min(1, (state.x - 1) / 3.5));
    playPhase = Math.asin(ratio);
    playRAF = requestAnimationFrame(animatePlay);
  }

  xSlider.addEventListener("input", function () {
    stopPlay();
    setX(parseFloat(xSlider.value), false);
  });

  vertexBtn.addEventListener("click", function () {
    stopPlay();
    setX(1, false);
  });

  playBtn.addEventListener("click", function () {
    if (state.playing) {
      stopPlay();
    } else {
      startPlay();
    }
  });

  resetBtn.addEventListener("click", function () {
    stopPlay();
    state.step = 0;
    state.x = 2;
    render();
  });

  document.querySelectorAll(".step-btn").forEach(function (btn) {
    btn.addEventListener("click", function () {
      stopPlay();
      state.step = Number(btn.dataset.step);
      render();
    });
  });

  drawStatic();
  render();
})();
