      flo = fm;
    }
  }
  return (lo + hi) / 2;
}

function frame(cfg){
  const L = 76, R = 716, T = 44, B = 472;
  const W = R - L, H = B - T;
  const { xMin, xMax, yMin, yMax } = cfg;
  const X = x => L + (x - xMin) / (xMax - xMin) * W;
  const Y = y => T + (yMax - y) / (yMax - yMin) * H;
  const xTicks = cfg.xTicks || makeTicks(xMin, xMax, 7);
  const yTicks = cfg.yTicks || makeTicks(yMin, yMax, 6);

  let s = '';
  s += `<rect x="0" y="0" width="760" height="520" fill="#ffffff"/>`;
  s += `<defs><clipPath id="plotClip"><rect x="${L}" y="${T}" width="${W}" height="${H}" rx="4"/></clipPath></defs>`;
  s += `<rect x="${L}" y="${T}" width="${W}" height="${H}" fill="#fbfdff"/>`;

  for (const x of xTicks) {
    const px = X(x);
    if (px < L - 2 || px > R + 2) continue;
    s += lineEl(px, T, px, B, '#e6edf6', 1);
    s += txt(px, B + 18, fmt(x, 2), '#64748b', { size: 11, anchor: 'middle', stroke: false });
  }
  for (const y of yTicks) {
    const py = Y(y);
    if (py < T - 2 || py > B + 2) continue;
    s += lineEl(L, py, R, py, '#e6edf6', 1);
    s += txt(L - 10, py + 4, fmt(y, 2), '#64748b', { size: 11, anchor: 'end', stroke: false });
  }

  s += `<rect x="${L}" y="${T}" width="${W}" height="${H}" fill="none" stroke="#cbd5e1" stroke-width="1.2"/>`;

  if (xMin <= 0 && 0 <= xMax) {
    s += lineEl(X(0), T, X(0), B, '#94a3b8', 1.5);
  }
  if (yMin <= 0 && 0 <= yMax) {
    s += lineEl(L, Y(0), R, Y(0), '#94a3b8', 1.5);
  }

  return { svg: s, X, Y, L, R, T, B, W, H };
}

  if (state.step <= 1) {
    state.a1 = clamp(state.a1, 0.2, 1.8);
    data = renderPart1(state.step);
  } else if (state.step <= 3) {
    state.a2 = clamp(state.a2, 0.2, 5);
    data = renderPart2(state.step);
  } else {
    state.r = clamp(state.r, 1.02, 8);
    data = renderLemma();
  }
  svgEl.innerHTML = data.svg;
  sideEl.innerHTML = data.side;
  syncControls();
  updateAnswers();
}
