      // 主三角形（前 2 s 或到当前时刻）
      if(showArea){
        const endMain = showLast ? Math.min(t, LAST_T) : t;
        const pts = underCurvePoints(0, endMain);
        if(pts){
          els.areaMain.setAttribute("points", pts);
          els.areaMain.style.display = "";
        } else {
          els.areaMain.style.display = "none";
        }
      } else {
        els.areaMain.style.display = "none";
      }

      // 最后 4 s 的三角形轮廓（2 s → 6 s）
      if(showLast){
        els.lastRegion.setAttribute("points",
          `${chartX(LAST_T)},${CH.bottom} ${chartX(LAST_T)},${chartY(velAt(LAST_T))} ${chartX(T_STOP)},${CH.bottom}`);
        els.lastRegion.style.display = "";
        els.t2Mark.style.display = "";
      } else {
        els.lastRegion.style.display = "none";
        els.t2Mark.style.display = "none";
      }

      // 黄色已扫过部分
      if(showArea && showLast && t > LAST_T + 0.001){
        els.areaLast.setAttribute("points", underCurvePoints(LAST_T, t));
        els.areaLast.style.display = "";
      } else {
        els.areaLast.style.display = "none";
      }

      // 面积数值标签
      const showTotalLabel = state.step === 2 && t >= T_STOP - 0.005;
      els.areaLabelTotal.style.display = showTotalLabel ? "" : "none";

      const showFirstPart = state.showLast && t >= LAST_T - 0.001;
      els.areaLabel1.style.display = showFirstPart ? "" : "none";
      els.areaLabel1.setAttribute("x", chartX(0.9));
      els.areaLabel1.setAttribute("y", 205);

      const showLastPart = state.step === 4 && t >= T_STOP - 0.005;
      els.areaLabel2.style.display = showLastPart ? "" : "none";
      els.areaLabel2.setAttribute("x", chartX(4));
      els.areaLabel2.setAttribute("y", 218);
    }

    function updateChart(t){
      const v = velAt(t);
      const px = chartX(t);
      const py = chartY(v);

      // 完整直线
      els.vtFull.setAttribute("x1", chartX(0));
      els.vtFull.setAttribute("y1", chartY(V0));
      els.vtFull.setAttribute("x2", chartX(T_STOP));
      els.vtFull.setAttribute("y2", chartY(0));

      // 已走过部分
      if(t > 0.02){
        els.vtProgress.style.display = "";
        els.vtProgress.setAttribute("x1", chartX(0));
        els.vtProgress.setAttribute("y1", chartY(V0));
        els.vtProgress.setAttribute("x2", px);
        els.vtProgress.setAttribute("y2", py);
      } else {
        els.vtProgress.style.display = "none";
      }

      // 当前点
      els.point.setAttribute("cx", px);
      els.point.setAttribute("cy", py);

      els.vGuide.setAttribute("x1", px);
      els.vGuide.setAttribute("x2", px);
      els.vGuide.setAttribute("y1", py);
      els.vGuide.setAttribute("y2", CH.bottom);

      els.hGuide.setAttribute("x1", CH.left);
      els.hGuide.setAttribute("x2", px);
      els.hGuide.setAttribute("y1", py);
      els.hGuide.setAttribute("y2", py);

      // 当前点标签：接近 0 时隐藏，避免与“t = 6 s”重叠
      if(v < 1.5){
        els.pointLabel.style.display = "none";
      } else {
        els.pointLabel.style.display = "";
        let lx = px + 10;
        let ly = py - 10;
        let anchor = "start";
        if(px > 340){ lx = px - 10; anchor = "end"; }
        if(py < 62){ ly = py + 22; }
        els.pointLabel.setAttribute("x", lx);
        els.pointLabel.setAttribute("y", ly);
        els.pointLabel.setAttribute("text-anchor", anchor);
        els.pointLabel.textContent = `v=${v.toFixed(1)}`;
      }

      // 停稳标记
      els.stopMark.style.display = state.step >= 1 ? "" : "none";

      updateAreas(t);
    }

    /* ---------- 跑道场景 ---------- */
    function updateScene(t){
      const v = velAt(t);
      const planeX = scenePos(posAt(t));

      els.scenePlane.setAttribute("transform", `translate(${planeX} ${SCENE_Y})`);

      const arrowStart = planeX - 18;
      if(v > 0.05){
        const arrowLen = Math.max(6, v * 2.25);
        els.velArrow.style.display = "";
        els.velArrow.setAttribute("x1", arrowStart);
        els.velArrow.setAttribute("x2", arrowStart + arrowLen);
        els.velLabel.style.display = "";
        els.velLabel.setAttribute("x", arrowStart);
        els.velLabel.setAttribute("text-anchor", "start");
        els.velLabel.textContent = `v = ${v.toFixed(1)} m/s`;
        els.accGroup.style.display = "";
        els.accGroup.setAttribute("transform", `translate(${arrowStart} 180)`);
      } else {
        els.velArrow.style.display = "none";
        els.velLabel.style.display = "";
        els.velLabel.setAttribute("x", planeX - 8);
        els.velLabel.setAttribute("text-anchor", "end");
        els.velLabel.textContent = "v = 0（已停稳）";
        els.accGroup.style.display = "none";
      }

      const stopVisible = state.step >= 2;
      els.stopLine.style.display = stopVisible ? "" : "none";
      els.stopLabel.style.display = stopVisible ? "" : "none";

      if(state.showLast){
        const lastStartSceneX = scenePos(LAST_X);
        const stopX = SCENE_X1;
        els.lastZoneFull.style.display = "";
        els.lastZoneFull.setAttribute("x", lastStartSceneX);
        els.lastZoneFull.setAttribute("width", stopX - lastStartSceneX);

        const doneWidth = Math.max(0, Math.min(planeX, stopX) - lastStartSceneX);
        els.lastZoneDone.style.display = "";
        els.lastZoneDone.setAttribute("x", lastStartSceneX);
        els.lastZoneDone.setAttribute("width", doneWidth);

        els.lastStartTick.style.display = "";
        els.lastStartTick.setAttribute("transform", `translate(${lastStartSceneX} 0)`);

        els.lastSceneBadge.style.display = "";
        els.lastSceneBadge.setAttribute("transform", `translate(${(lastStartSceneX + stopX) / 2} 0)`);

        if(state.step === 4 && t >= T_STOP - 0.005){
          els.lastSceneBadgeText.textContent = "最后 4 s：48 m";
        } else {
          els.lastSceneBadgeText.textContent = "最后 4 s 区间";
        }
      } else {
        els.lastZoneFull.style.display = "none";
        els.lastZoneDone.style.display = "none";
        els.lastStartTick.style.display = "none";
        els.lastSceneBadge.style.display = "none";
      }
    }

    /* ---------- 时间轴 ---------- */
    function updateTimeline(t){
      const tx = timeX(t);
      els.timeMarker.setAttribute("x1", tx);
      els.timeMarker.setAttribute("x2", tx);
      els.timeDot.setAttribute("cx", tx);

      if(state.showLast){
        const a = timeX(LAST_T), b = timeX(T_STOP);
        els.lastTimeZone.style.display = "";
        els.lastTimeZone.setAttribute("x", a);
        els.lastTimeZone.setAttribute("width", b - a);
        els.lastBracket.style.display = "";
        els.bracketLine.setAttribute("x1", a);
        els.bracketLine.setAttribute("x2", b);
        els.bracketTickL.setAttribute("x1", a);
        els.bracketTickL.setAttribute("x2", a);
        els.bracketTickR.setAttribute("x1", b);
        els.bracketTickR.setAttribute("x2", b);
        els.bracketText.setAttribute("x", (a + b) / 2);
      } else {
        els.lastTimeZone.style.display = "none";
        els.lastBracket.style.display = "none";
      }
    }

    /* ---------- 读数 ---------- */
    function updateReadouts(t){
      const v = velAt(t);
      const x = posAt(t);
      els.readT.textContent = t.toFixed(1);
      els.readV.textContent = v.toFixed(1);
      els.readX.textContent = x.toFixed(1);
      els.statusPill.textContent = v <= 0.01 ? "已停稳" : `t = ${t.toFixed(1)} s`;

      if(state.showLast){
        const within = Math.max(0, x - LAST_X);
        const isFinal = state.step === 4 && t >= T_STOP - 0.005;
        els.lastReadout.style.display = "";
        els.lastReadout.classList.toggle("final", isFinal);
        if(isFinal){
          els.lastReadout.innerHTML = `最后 4 s 内位移：<strong>${within.toFixed(1)} m</strong>`;
        } else {
          els.lastReadout.innerHTML = `最后 4 s 区间内已滑行：<strong>${within.toFixed(1)} m</strong>`;
        }
      } else {
        els.lastReadout.style.display = "none";
        els.lastReadout.innerHTML = "";
      }
    }

    function renderAll(){
      els.timeSlider.value = String(state.t);
      updateScene(state.t);
      updateTimeline(state.t);
      updateChart(state.t);
      updateReadouts(state.t);
    }

    /* ---------- 步骤切换 ---------- */
    function setStep(step){
      state.playing = false;
      state.step = step;
      state.showLast = step >= 3;
      const targets = [0, 6, 6, 2, 6];
      state.t = targets[step] !== undefined ? targets[step] : 0;
      updateStepButtons();
      updateAnswers();
      renderFormula();
      updatePlayButton();
      renderAll();
    }

    /* ---------- 播放 ---------- */
    function updatePlayButton(){
      els.playBtn.textContent = state.playing ? "暂停" : "播放";
      els.playBtn.classList.toggle("primary", !state.playing);
    }

    function playTo(endT, duration){
      endT = clampT(endT);
      const startT = state.t;
      if(Math.abs(endT - startT) < 0.001){
        state.playing = false;
        updatePlayButton();
        renderAll();
        return;
      }
      state.playing = true;
      state.animStart = performance.now();
      state.animFrom = startT;
      state.animTo = endT;
      state.animDur = duration || Math.max(650, Math.abs(endT - startT) * 380);
      updatePlayButton();
      requestAnimationFrame(animTick);
    }

    function animTick(now){
      if(!state.playing) return;
      const p = Math.min(1, (now - state.animStart) / state.animDur);
      state.t = state.animFrom + (state.animTo - state.animFrom) * p;
      if(p >= 1){
        state.t = state.animTo;
        state.playing = false;
        updatePlayButton();
      }
      renderAll();
      if(state.playing){
        requestAnimationFrame(animTick);
      }
    }

    function togglePlay(){
      if(state.playing){
        state.playing = false;
        updatePlayButton();
        return;
      }
      if(state.t >= VIEW_MAX - 0.005){
        state.t = 0;
      }
      playTo(VIEW_MAX, Math.max(700, (VIEW_MAX - state.t) * 380));
    }

    /* ---------- 事件绑定 ---------- */
    els.stepBtns.forEach(btn => {
      btn.addEventListener("click", () => setStep(Number(btn.dataset.step)));
    });

    els.timeSlider.addEventListener("input", (e) => {
      state.playing = false;
      state.t = clampT(parseFloat(e.target.value));
      updatePlayButton();
      renderAll();
    });

    // 时间轴直接拖动
    let draggingTimeline = false;
    function tFromPointer(evt){
      const rect = els.timeline.getBoundingClientRect();
      if(rect.width <= 0) return state.t;
      const svgX = (evt.clientX - rect.left) * (960 / rect.width);
      const t = ((svgX - TIME_X0) / (TIME_X1 - TIME_X0)) * VIEW_MAX;
      return clampT(t);
    }
    els.timeline.addEventListener("pointerdown", (e) => {
      draggingTimeline = true;
      state.playing = false;
      updatePlayButton();
      state.t = tFromPointer(e);
      renderAll();
      if(els.timeline.setPointerCapture){
        try{ els.timeline.setPointerCapture(e.pointerId); }catch(err){}
      }
      e.preventDefault();
    });
    els.timeline.addEventListener("pointermove", (e) => {
      if(!draggingTimeline) return;
      state.t = tFromPointer(e);
      renderAll();
    });
    function endDrag(){ draggingTimeline = false; }
    els.timeline.addEventListener("pointerup", endDrag);
    els.timeline.addEventListener("pointercancel", endDrag);
    els.timeline.addEventListener("pointerleave", (e) => {
      if(e.buttons === 0) endDrag();
    });

    els.playBtn.addEventListener("click", togglePlay);
    els.resetBtn.addEventListener("click", () => setStep(0));

    els.formulaCard.addEventListener("click", (e) => {
      const btn = e.target.closest("[data-demo]");
      if(!btn) return;
      const parts = btn.dataset.demo.split(",").map(Number);
      state.playing = false;
      state.t = clampT(parts[0]);
      renderAll();
      playTo(parts[1], Math.max(750, Math.abs(parts[1] - parts[0]) * 380));
    });

    /* ---------- 初始化 ---------- */
    buildTimeTicks();
    buildChartTicks();
    setStep(0);
  })();
</script>
</body>
