## 五维评分（1–5）
- 一致性：是否保留原题对象、条件、各小问和允许范围。
- 正确性：推导、计算、符号、边界、结论和图中数值是否正确。
- 页面效果：桌面显示是否清楚，图形、文字与公式有无重叠、裁切。此次不评分手机适配。
- 交互效果：控制量来历和范围是否合理，图形、公式、数值是否同步，反馈是否靠近操作。无需交互的题可凭恰当静态表达得高分。
- 教学效果：代入具备相应高中基础、刚读完题但不会做本题的学生，检查能否通过可视化循序渐进地理解条件转化、每步依据和答案由来。不能用评审者已知的解法、源码或参考解析补齐页面缺失的解释。关注学生哪里会卡住、信息量和理解难度是否合理，以及学完能否说明为什么这样解。

教学效果专项锚点：1=基本学不到解法或受到核心误导；2=关键步骤需要预先会做题或外部讲解；3=主线可跟随，但存在重要解释不足或明显理解负担；4=关键转化与图形对应清楚，仅有局部障碍；5=从题设到答案循序渐进，关键关系都有可见依据，无需预先会做本题即可理解并复述。关键步骤必须靠评审者补全时最高2分。此评分是基于页面证据的判断，不代表已经验证真实学生的学习效果。

统一锚点：1=严重缺陷，基本不达标；2=有明显问题，需要较大修改；3=基本可用，仍有明显不足；4=良好，仅少量局部问题；5=完整清晰，该维度未发现实质问题。
## 综合评分（0–5，独立填写，不自动平均）

重点检查：

5. 学完是否有依据认为能独立复述：学生能否说明“为什么这样做、每一步得到什么、答案为何成立”，而不仅是记住最后选项或数值。这只是基于页面证据的教学判断，不得声称已经证明真实学生学会了。

  function renderGap(s){
    const W = 460, H = 245, L = 48, R = 16, T = 22, B = 34;
    const tmax = state.mode === "case1" ? 14 : 10;

    const vals = [];
    for (let i = 0; i <= 240; i++) {
      const t = tmax * i / 240;
      vals.push(gapAt(t, state.mode, state.x0));
    }

    let mn = Math.min.apply(null, vals);
    let mx = Math.max.apply(null, vals);
    let pad = Math.max(4, (mx - mn) * 0.16);
    let rawMin = Math.min(0, mn - pad);
    let rawMax = Math.max(0, mx + pad);
    if (rawMax - rawMin < 8) {
      rawMin -= 4;
      rawMax += 4;
    }
    const step = niceStep((rawMax - rawMin) / 4);
    const ymin = Math.floor(rawMin / step) * step;
    const ymax = Math.ceil(rawMax / step) * step;

    const xp = function(t){ return L + (t / tmax) * (W - L - R); };
    const yp = function(v){ return T + (ymax - v) / (ymax - ymin) * (H - T - B); };

    let g = '<rect x="0" y="0" width="460" height="245" rx="10" fill="#ffffff"></rect>';

    for (let t = 0; t <= tmax + 1e-9; t += 2) {
      const x = xp(t);
      g += '<line x1="' + x.toFixed(1) + '" y1="' + T + '" x2="' + x.toFixed(1) + '" y2="' + (H - B) + '" stroke="' + COLORS.grid + '" stroke-width="1"></line>';
      g += '<text x="' + x.toFixed(1) + '" y="' + (H - 12) + '" text-anchor="middle" font-size="10" fill="#7d8da3">' + t + '</text>';
    }

    const vCount = Math.min(20, Math.round((ymax - ymin) / step));
    for (let i = 0; i <= vCount; i++) {
      const v = ymin + i * step;
      const y = yp(v);
      g += '<line x1="' + L + '" y1="' + y.toFixed(1) + '" x2="' + (W - R) + '" y2="' + y.toFixed(1) + '" stroke="' + COLORS.grid + '" stroke-width="1"></line>';
      g += '<text x="' + (L - 6) + '" y="' + (y + 3.5).toFixed(1) + '" text-anchor="end" font-size="10" fill="#7d8da3">' + labelNum(v, step) + '</text>';
    }

    g += '<line x1="' + L + '" y1="' + (H - B) + '" x2="' + (W - R) + '" y2="' + (H - B) + '" stroke="' + COLORS.axis + '" stroke-width="1.2"></line>';
    g += '<line x1="' + L + '" y1="' + T + '" x2="' + L + '" y2="' + (H - B) + '" stroke="' + COLORS.axis + '" stroke-width="1.2"></line>';
    g += '<text x="' + (W - R) + '" y="' + (H - 12) + '" text-anchor="end" font-size="10.5" fill="#70839b">t / s</text>';

    const y0 = yp(0);
    g += '<line x1="' + L + '" y1="' + y0.toFixed(1) + '" x2="' + (W - R) + '" y2="' + y0.toFixed(1) + '" stroke="' + COLORS.zero + '" stroke-width="1.4" opacity="0.58"></line>';
    g += '<text x="' + (W - R) + '" y="' + (y0 - 5).toFixed(1) + '" text-anchor="end" font-size="9.5" fill="#c45767">0 轴：相遇</text>';

    let pts = [];
    for (let i = 0; i <= 240; i++) {
      const t = tmax * i / 240;
      const v = gapAt(t, state.mode, state.x0);
      pts.push(xp(t).toFixed(1) + "," + yp(v).toFixed(1));
    }
    const lineColor = state.mode === "case1" ? "#355d8f" : "#4b6f97";
    g += '<polyline points="' + pts.join(" ") + '" fill="none" stroke="' + lineColor + '" stroke-width="3" stroke-linecap="round" stroke-linejoin="round"></polyline>';

    let roots = [];
    if (state.mode === "case1") {
      if (state.revealed.meet) roots = [12];
    } else {
      roots = rootInfo(state.x0).roots;
    }

    roots.forEach(function(rt, idx){
      if (rt >= 0 && rt <= tmax) {
        const x = xp(rt);
        g += '<line x1="' + x.toFixed(1) + '" y1="' + T + '" x2="' + x.toFixed(1) + '" y2="' + (H - B) + '" stroke="#df5b68" stroke-width="1.2" stroke-dasharray="3 4" opacity="0.55"></line>';
        g += '<circle cx="' + x.toFixed(1) + '" cy="' + y0.toFixed(1) + '" r="4.6" fill="#ffffff" stroke="#df5b68" stroke-width="2.2"></circle>';
        if (state.mode === "case3" || state.revealed.meet) {
          g += '<text x="' + x.toFixed(1) + '" y="' + (H - B + 16) + '" text-anchor="middle" font-size="10.5" font-weight="800" fill="#b64858">t' + (idx + 1) + '</text>';
        }
      }
    });

    if (state.mode === "case1" && state.revealed.max) {
      const x4 = xp(4);
      const yv = yp(gapAt(4, "case1", state.x0));
      g += '<line x1="' + x4.toFixed(1) + '" y1="' + yv.toFixed(1) + '" x2="' + x4.toFixed(1) + '" y2="' + y0.toFixed(1) + '" stroke="#e09139" stroke-width="1.3" stroke-dasharray="3 4"></line>';
      g += '<circle cx="' + x4.toFixed(1) + '" cy="' + yv.toFixed(1) + '" r="5" fill="#ffffff" stroke="#e09139" stroke-width="2.2"></circle>';
      g += '<text x="' + (x4 + 6).toFixed(1) + '" y="' + (yv - 7).toFixed(1) + '" font-size="10.5" font-weight="800" fill="#b87522">最大 80 m（vA=vB）</text>';

    if (state.mode === "case3" && (state.focus === "min" || state.revealed.cond)) {
      const x4 = xp(4);
      const minV = state.x0 - 24;
      const yv = yp(minV);
      g += '<line x1="' + x4.toFixed(1) + '" y1="' + yv.toFixed(1) + '" x2="' + x4.toFixed(1) + '" y2="' + y0.toFixed(1) + '" stroke="#d78a2e" stroke-width="1.3" stroke-dasharray="3 4" opacity="0.8"></line>';
      g += '<circle cx="' + x4.toFixed(1) + '" cy="' + yv.toFixed(1) + '" r="5" fill="#ffffff" stroke="#d78a2e" stroke-width="2.2"></circle>';

    }

    const cx = xp(s.t);
    const cy = yp(gapAt(s.t, state.mode, state.x0));
    g += '<line x1="' + cx.toFixed(1) + '" y1="' + T + '" x2="' + cx.toFixed(1) + '" y2="' + (H - B) + '" stroke="#203a5c" stroke-width="1.4" stroke-dasharray="3 4" opacity="0.58"></line>';
    g += '<circle cx="' + cx.toFixed(1) + '" cy="' + cy.toFixed(1) + '" r="4.8" fill="' + (state.mode === "case1" ? "#355d8f" : "#4b6f97") + '" stroke="#ffffff" stroke-width="1.8"></circle>';

    const metricText = state.mode === "case1" ? "S = xA − xB" : "D = xB − xA";
    const axisNote = state.mode === "case1" ? "纵轴 S / m；间距 = |S|" : "纵轴 D / m；间距 = |D|";
    g += '<text x="' + (L + 5) + '" y="14" font-size="10.5" font-weight="700" fill="#72859d">' + metricText + '</text>';
    g += '<text x="' + (W - R) + '" y="14" text-anchor="end" font-size="9.5" font-weight="700" fill="#72859d">' + axisNote + '</text>';

    dom.gapSvg.innerHTML = g;
    dom.gapFormulaLabel.textContent = metricText + "（间距 = |" + (state.mode === "case1" ? "S" : "D") + "|）";
  }

  function getStatus(s){
    const EPS = 0.02;
    if (state.mode === "case1") {
      const t = s.t;
      if (t < 0.02) return ["初态", "A 在前、B 在后，相距 56.0 m；A 相对 B 更快，间距先增大。"];
      if (t < 3.98) return ["运动中", "vA > vB，间距继续增大；当前间距 " + fmt(s.gap, 1) + " m。"];
      if (Math.abs(t - 4) <= EPS) return ["峰值", "vA = vB = 8 m/s；相对速度为 0，间距达到最大 80 m。"];
      if (t < 7.98) return ["运动中", "vA < vB，间距开始缩小；当前间距 " + fmt(s.gap, 1) + " m。"];
      if (Math.abs(t - 8) <= EPS) return ["关键点", "t = 8 s：A 停下（x=120 m），B 在 64 m，两车相距 56.0 m。"];
      if (t < 11.98) {
        if (s.gap < 6) return ["接近相遇", "B 正在追上 A；实时间距仅 " + fmt(s.gap, 1) + " m。"];
        return ["追赶", "A 静止，B 继续加速追赶；当前间距 " + fmt(s.gap, 1) + " m。"];
      }
      if (Math.abs(t - 12) <= EPS) return ["相遇", "t = 12 s 时，B 追上 A；两车位置同为 120 m，间距 0 m。"];
      return ["已相遇", "B 已在 A 前，间距重新增大；实时间距 " + fmt(s.gap, 1) + " m。"];
    }

    const info = rootInfo(state.x0);
    const t = s.t;
    const x0 = state.x0;
    if (x0 < 24 - 1e-9) {
      const t1 = info.roots[0];
      const t2 = info.roots[1];
      if (t < t1 - EPS) return ["追近", "A 在后且更快（vA>vB），间距缩小；实时间距 " + fmt(s.gap, 1) + " m。"];
      if (Math.abs(t - t1) <= EPS) return ["第一次相遇", "t ≈ " + fmt(t1, 1) + " s，A 追上 B；此后 A 在前，D<0。"];
      if (t < 4 - EPS) return ["A 在前，间距增大", "第一次相遇后 vA 仍大于 vB，A 继续拉开；间距 = |D| = " + fmt(s.gap, 1) + " m。"];
      if (Math.abs(t - 4) <= EPS) return ["间距转折", "t = 4 s：vA = vB，A 不再拉开；D_min = x₀−24 = " + fmt(x0 - 24, 1) + " m，|D| = " + fmt(Math.abs(x0 - 24), 1) + " m。"];
      if (t < t2 - EPS) return ["B 开始追近", "t > 4 s 后 vB > vA，B 逐渐追近；D 仍小于 0，间距 = |D| = " + fmt(s.gap, 1) + " m。"];
      if (Math.abs(t - t2) <= EPS) return ["第二次相遇", "t ≈ " + fmt(t2, 1) + " s，B 再次追上 A；D = 0。"];
      return ["B 在前", "第二次相遇后 B 更快且在前，间距重新增大；实时间距 " + fmt(s.gap, 1) + " m。"];
    } else if (Math.abs(x0 - 24) < 1e-9) {
      if (t < 4 - EPS) return ["追近", "A 从后追近 B；实时间距 " + fmt(s.gap, 1) + " m。"];
      if (Math.abs(t - 4) <= EPS) return ["临界", "t = 4 s，两车恰好接触一次（D_min = 0），没有穿过彼此。"];
      return ["远离", "B 加速后在前，间距增大，不再相遇。"];
    } else {
      if (t < 4 - EPS) return ["追近", "A 追近 B；实时间距 " + fmt(s.gap, 1) + " m。"];
      if (Math.abs(t - 4) <= EPS) return ["未追上", "t = 4 s 时，间距最小：x₀−24 = " + fmt(x0 - 24, 1) + " m > 0，仍未追上。"];
      return ["远离", "B 更快且在前，两车间距增大；实时间距 " + fmt(s.gap, 1) + " m。"];
    }
  }

    let html = "";

      '<div class="result-label">当前 x₀ = ' + fmt(state.x0, 1) + ' m 时</div>' +
      '<div class="result-value">' + countText + '</div>' +
    '</div>';

    html += '<div class="result-row">' +
      '<div class="result-label">（3）两车可以相遇两次的条件</div>' +
      '<div class="result-value ' + (state.revealed.cond ? "" : "pending") + '">' +
        (state.revealed.cond ? "0 < x₀ < 24 m" : "拖动 x₀ 观察交点，再点“归纳条件”") +
      '</div>' +
      (state.revealed.cond ? '<div class="formula-mini">先合并位置：A 的位移 xA = 16t − t²（t≤8 s；之后停在 64 m），B 的位置 xB = x₀ + 4t + ½t²。<br>取 D = xB − xA = x₀ − 12t + 1.5t²，两车间距 = |D|。<br>t = 4 s 时 D 取最小值 D_min = x₀ − 24。<br>要相遇两次：D 必须穿过 0（D_min < 0），且初始 x₀ > 0。<br>所以：0 < x₀ < 24 m。</div>' : '') +
    '</div>';

    dom.conclusions.innerHTML = html;
  }

  function render(){
    const maxT = state.mode === "case1" ? 14 : 10;
    if (state.t > maxT) state.t = maxT;
    const s = motion(state.t, state.mode, state.x0);
    updateTimeUI(s);
    renderScene(s);
    renderVT(s);
    renderGap(s);
    updateReadouts(s);
    updateStatus(s);
    updateStepButtons();
    updateStepNote();
    updateExploration();
    updateConclusions(s);
  }

  function stopAnimation(){
    state.playing = false;
    if (rafId !== null) {
      cancelAnimationFrame(rafId);
      rafId = null;
    }
    lastTs = 0;
  }

  function playLoop(ts){
    if (!state.playing) return;
    if (!lastTs) lastTs = ts;
    const dt = Math.min(0.06, (ts - lastTs) / 1000);
    lastTs = ts;
    const maxT = state.mode === "case1" ? 14 : 10;
    state.t = Math.min(maxT, state.t + dt);
    if (state.t >= maxT - 1e-7) {
      state.t = maxT;
      state.playing = false;
    }
    render();
    if (state.playing) {
      rafId = requestAnimationFrame(playLoop);
    }
  }

  function togglePlay(){
    if (state.playing) {
      stopAnimation();
      render();
      return;
    }
    const maxT = state.mode === "case1" ? 14 : 10;
    if (state.t >= maxT - 1e-6) state.t = 0;
    state.playing = true;
    lastTs = 0;
    rafId = requestAnimationFrame(playLoop);
    render();
  }

  function setMode(mode){
    state.mode = mode;
    state.t = 0;
    state.focus = "none";
    stopAnimation();
    state.revealed = initialRevealed();
    if (dom.sceneSubtitle) {
      dom.sceneSubtitle.textContent = mode === "case1"
        ? "同向直公路：A 在前、B 在后，相距 56 m；A 减速，B 加速。"
        : "同向直公路：A 在后、B 在前，初始间距 x₀ 可调；A 减速，B 加速。";
    }
    dom.modeCase1.classList.toggle("active", mode === "case1");
    dom.modeCase3.classList.toggle("active", mode === "case3");
    dom.case1Panel.classList.toggle("hidden", mode !== "case1");
    dom.case3Panel.classList.toggle("hidden", mode !== "case3");
    dom.timeSlider.max = mode === "case1" ? 14 : 10;
    dom.timeSlider.value = 0;
    render();
  }

  function bindEvents(){
    dom.modeCase1.addEventListener("click", function(){ setMode("case1"); });
    dom.modeCase3.addEventListener("click", function(){ setMode("case3"); });

    dom.playBtn.addEventListener("click", togglePlay);

    dom.timeSlider.addEventListener("input", function(){
      stopAnimation();
      const maxT = state.mode === "case1" ? 14 : 10;
      state.t = clamp(parseFloat(dom.timeSlider.value), 0, maxT);
      state.focus = "free";
      render();
    });

    document.querySelectorAll("[data-dt]").forEach(function(btn){
      btn.addEventListener("click", function(){
        stopAnimation();
        const maxT = state.mode === "case1" ? 14 : 10;
        state.t = clamp(state.t + parseFloat(btn.dataset.dt), 0, maxT);
        state.focus = "free";
        render();
      });
    });

    dom.resetBtn.addEventListener("click", function(){
      stopAnimation();
      state.t = 0;
      state.focus = "none";
      state.revealed = initialRevealed();
      render();
    });

    document.querySelectorAll(".step-btn").forEach(function(btn){
      btn.addEventListener("click", function(){
        stopAnimation();
        const step = btn.dataset.step;
        state.focus = step;
        if (step === "start") {
          state.t = 0;
          state.revealed = initialRevealed();
        }
        if (step === "max") {
          state.t = 4;
          state.revealed.max = true;
        }
        if (step === "stop") {
          state.t = 8;
          state.revealed.stop = true;
        }
        if (step === "meet") {
          state.t = 12;
          state.revealed.stop = true;
          state.revealed.meet = true;
        }
        if (step === "min") {
          state.t = 4;
        }
        render();
      });
    });

    dom.x0Slider.addEventListener("input", function(){
      stopAnimation();
      state.x0 = parseFloat(dom.x0Slider.value);
      state.focus = "free";
      render();
    });

    document.querySelectorAll("[data-x0]").forEach(function(btn){
      btn.addEventListener("click", function(){
        stopAnimation();
        state.x0 = parseFloat(btn.dataset.x0);
        state.t = 4;
        state.focus = "min";
        render();
      });
    });

    dom.condBtn.addEventListener("click", function(){
      stopAnimation();
      state.revealed.cond = true;
      state.focus = "cond";
      state.t = 4;
      render();
    });
  }

  initTicks();
  bindEvents();
  dom.case1Panel.classList.remove("hidden");
  dom.case3Panel.classList.add("hidden");
  render();
})();
