const NS = 'http://www.w3.org/2000/svg';

function hexPoints(cx, cy, r, rot=-90){
  const pts = [];
  for(let i=0;i<6;i++){
    const a = (rot + i*60) * Math.PI/180;
    pts.push([cx + r*Math.cos(a), cy + r*Math.sin(a)]);
  }
  return pts;
}
function benzenePath(cx, cy, r, rot=-90){
  const pts = hexPoints(cx,cy,r,rot);
  return 'M' + pts.map(p=>p.join(',')).join('L') + 'Z';
}
function benzene(cx, cy, r, rot=-90, stroke='#334155'){
  const inner = benzenePath(cx, cy, r*0.70, rot);
  return `<path d="${benzenePath(cx,cy,r,rot)}" fill="#fff" stroke="${stroke}" stroke-width="1.8"/>` +
         `<path d="${inner}" fill="none" stroke="${stroke}" stroke-width="1" stroke-dasharray="2 4"/>`;
}
function line(x1,y1,x2,y2,color='#334155',w=1.8){
  return `<line x1="${x1}" y1="${y1}" x2="${x2}" y2="${y2}" stroke="${color}" stroke-width="${w}" stroke-linecap="round"/>`;
}
function txt(x,y,str,color='#1e293b',size=13,anchor='middle',weight='normal'){
  return `<text x="${x}" y="${y}" fill="${color}" font-size="${size}" text-anchor="${anchor}" font-weight="${weight}" font-family="system-ui,-apple-system,'Microsoft YaHei',sans-serif">${str}</text>`;
}

let playTimer = null;

function renderRouteRow(id, keys){
  const row = document.getElementById(id);
  if(!row) return;
  row.innerHTML = '';
  keys.forEach((k, idx)=>{
    const node = document.createElement('div');
    node.className = 'route-node';
    node.dataset.key = k;
    node.innerHTML = `<div class="node-mini">${miniSvg(k)}</div><div class="node-label">${k}</div>`;
    node.addEventListener('click', ()=> selectNode(k));
    row.appendChild(node);
    if(idx < keys.length-1){
      const arrow = document.createElement('div');
      arrow.className = 'route-arrow';
      arrow.textContent = '→';
      row.appendChild(arrow);
    }
  });
  if(id === 'route-row-2'){
    const extra = document.createElement('div');
    extra.className = 'route-extra';
    extra.textContent = 'G 与 C 汇合生成 H';
    row.appendChild(extra);
  }
}

function normalizeKey(c1,c2,c3){
  const a1 = c1.slice().sort();
  const a2 = c2.slice().sort();
  const a3 = c3.slice().sort();
  const a = JSON.stringify([a1,a2,a3]);
  const b = JSON.stringify([a3,a2,a1]);
  return a < b ? a : b;
}
isoStructures.forEach(s=>{
  s.key = normalizeKey(s.c1, s.c2, s.c3);
});

function renderChips(){
  document.querySelectorAll('#iso-chips .chip').forEach(btn=>{
    const id = +btn.dataset.chip;
    const chip = isoChips[id];
    btn.disabled = chip.used;
    btn.classList.toggle('used', chip.used);
    btn.classList.toggle('selected', selectedChip === id);
  });
}

function renderBuilder(){
  [1,2,3].forEach(c=>{
    const node = document.querySelector(`.carbon-node[data-c="${c}"]`);
    if(!node) return;
    const slots = node.querySelector('.slots');
    slots.innerHTML = '';
    isoPlaced[c].forEach(chipId=>{
      const chip = isoChips[chipId];
      const span = document.createElement('span');
      span.className = 'slot';
      span.textContent = chip.sub === 'OCH3' ? '—OCH₃' : '—OH';
      span.style.color = chip.sub === 'OH' ? '#dc2626' : '#7c3aed';
      span.addEventListener('click', (e)=>{
        e.stopPropagation();
        isoPlaced[c] = isoPlaced[c].filter(id=>id !== chipId);
        isoChips[chipId].used = false;
        renderBuilder();
      });
      slots.appendChild(span);
    });
  });
  renderChips();
}

function isoFeedback(msg, color){
  const fb = document.getElementById('iso-feedback');
  fb.innerHTML = color ? `<span style="color:${color}">${msg}</span>` : msg;
}

function getCurrentKey(){
  const subs = [1,2,3].map(c=> isoPlaced[c].map(id=> isoChips[id].sub));
  return normalizeKey(subs[0], subs[1], subs[2]);
}

function appendFoundCard(s){
  const list = document.getElementById('iso-found');
  if(list.querySelector(`[data-sid="${s.id}"]`)) return;
  const card = document.createElement('div');
  card.className = 'iso-card' + (s.nmr ? ' nmr-hit' : '');
  card.dataset.sid = s.id;
  card.innerHTML = `
    <div class="iso-card-head">结构 ${s.id} ${s.nmr ? '<span class="badge">NMR 符合</span>' : ''}</div>
    <div class="iso-formula">${s.formula}</div>
    <div class="iso-cat">${s.cat}</div>
  `;
  card.addEventListener('click', ()=> showIsoDetail(s));
  list.appendChild(card);
}

function checkIsoComplete(){
  if(!isoChips.every(c=>c.used)) return;
  const key = getCurrentKey();
  const found = isoStructures.find(s=>s.key === key);
  if(found){
    if(!foundKeys.has(key)){
      foundKeys.add(key);
      appendFoundCard(found);
      isoFeedback(`合法结构：${found.formula}（${found.cat}）`, '#16a34a');
    }else{
      isoFeedback(`这个结构与已找到的「${found.formula}」相同。`, '#64748b');
    }
    if(found.nmr){
      isoFeedback(`合法结构：${found.formula}。它的氢谱为 4 组峰，面积比 6:3:2:1。`, '#16a34a');
    }
  }else{
    isoFeedback('这个组合不合法或不是题目条件下的结构，检查一下价数或位置。', '#dc2626');
  }
}

function drawSynth(stage){
  const svg = document.getElementById('svg-synth');
  const cx=150, cy=120, r=40;
  let s = '';

  // 苯环
  s += `<path d="${benzenePath(cx,cy,r,0)}" fill="#fff" stroke="#334155" stroke-width="2"/>`;
  // 侧链在 1 位（右侧，角度 0°）
  const p1 = [cx + r, cy];
  // 2 位（右下，角度 60°）Cl
  const p2 = [cx + r*Math.cos(60*Math.PI/180), cy + r*Math.sin(60*Math.PI/180)];
  // 4 位（左侧，角度 180°）Cl
  const p4 = [cx + r*Math.cos(180*Math.PI/180), cy + r*Math.sin(180*Math.PI/180)];

  s += line(p2[0], p2[1], p2[0]+14, p2[1]+10);
  s += txt(p2[0]+22, p2[1]+16, 'Cl', '#16a34a', 14, 'middle', '700');
  s += line(p4[0], p4[1], p4[0]-14, p4[1]);
  s += txt(p4[0]-24, p4[1]+5, 'Cl', '#16a34a', 14, 'middle', '700');

  if(stage === 0){
    s += line(p1[0], p1[1], p1[0]+28, p1[1]);
    s += txt(p1[0]+46, p1[1]+5, 'CH₂OH', '#dc2626', 14, 'middle', '700');

    s += txt(260, 40, '二氯苄醇', '#64748b', 13);

    s += txt(p1[0]+58, p1[1]-22, 'CH(OH)', '#dc2626', 13, 'middle', '700');
    s += line(p1[0]+58, p1[1]-18, p1[0]+80, p1[1]-8);
    s += txt(p1[0]+98, p1[1]-4, 'CH₂NO₂', '#dc2626', 13, 'middle', '700');

function updateSynthSteps(){
  const wrap = document.getElementById('synth-steps');
  const labels = {
    ox:'氧化',
    con:'与 CH₃NO₂ 缩合',
    red:'还原硝基'
  };
  wrap.innerHTML = '';
  synthSeq.slice(0, synthStage).forEach((step, i)=>{
    const pill = document.createElement('span');
    pill.className = 'synth-step-pill';
    pill.textContent = labels[step];
    wrap.appendChild(pill);
    if(i < synthStage-1){
      const arr = document.createElement('span');
      arr.className = 'synth-step-arrow';
      arr.textContent = '→';
      wrap.appendChild(arr);
    }
  });
}

function initSynth(){
  document.querySelectorAll('[data-step]').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      const step = btn.dataset.step;
      const fb = document.getElementById('synth-feedback');
      if(synthStage >= 3) return;

      if(synthSeq[synthStage] === step){
        btn.classList.add('done');
        synthStage++;
        drawSynth(synthStage);
        updateSynthSteps();

        if(synthStage === 1) fb.innerHTML = '苄醇被氧化为醛。下一步要引入硝基。';
        if(synthStage === 2) fb.innerHTML = '醛与 CH₃NO₂ 缩合，得到硝基醇。下一步还原硝基。';
        if(synthStage === 3){
          fb.innerHTML = '<span style="color:#16a34a">路线完成！</span> 苄醇 → 氧化 → 缩合 → 还原，得到目标氨基醇。';
          document.getElementById('conclusion-synth').innerHTML = '<div class="conclusion"><strong>合成路线：</strong>二氯苄醇 <span class="muted">—MnO₂→</span> 二氯苯甲醛 <span class="muted">+CH₃NO₂/碱→</span> 硝基醇 <span class="muted">—Fe/HCl→</span> 二氯苯基—CH(OH)—CH₂NH₂。</div>';
        }
      }else{
        btn.classList.add('shake');
        setTimeout(()=>btn.classList.remove('shake'), 380);
        fb.innerHTML = '<span style="color:#dc2626">顺序不对。</span> 先想想：苄醇要变成醛，才能与 CH₃NO₂ 缩合引入硝基。';
      }
    });
  });
}
