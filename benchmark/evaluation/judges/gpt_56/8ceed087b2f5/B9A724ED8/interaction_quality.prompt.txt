  lab.appendChild(mk("tspan", {}, "y = e"));

}

/* ==================== 步骤 1：第(1)问 面积比较 ==================== */
function drawStep1(a){
  const m = mapper(-0.25, 2.72, -0.9, 9.6);
  const C = x => Math.exp(x) - x*x/2 - 1;
  const L = x => a*x;
  const XMAX = 2.66;

  /* 色带 */
  const N = 320;
  let posP = "", negP = "";
  for(let i = 0; i <= N; i++){
    const x = XMAX * i / N;
    const yt = C(x), yb = L(x);
    if(!isFinite(yt) || !isFinite(yb)) continue;
    const px = m.X(x).toFixed(1);
    const seg = "M" + px + "," + m.Y(yt).toFixed(1) + "L" + px + "," + m.Y(yb).toFixed(1);
    if(yt >= yb) posP += seg; else negP += seg;
  }
  const wBand = (BOX.x1 - BOX.x0) / N * 1.9;
  if(negP) board.appendChild(mk("path", {d:negP, stroke:"rgba(220,38,38,0.42)",
    "stroke-width":wBand, fill:"none", "stroke-linecap":"butt"}));
  if(posP) board.appendChild(mk("path", {d:posP, stroke:"rgba(5,150,105,0.38)",
    "stroke-width":wBand, fill:"none", "stroke-linecap":"butt"}));

  board.appendChild(drawAxes(m, [0.5,1,1.5,2,2.5], [2,4,6,8]));

  board.appendChild(mk("path", {d: curvePath(m, C, 0, XMAX, 320), fill:"none",
    stroke:"#2563eb", "stroke-width":3.2, "stroke-linejoin":"round"}));
  board.appendChild(mk("path", {d: curvePath(m, L, 0, XMAX, 2), fill:"none",
    stroke:"#d97706", "stroke-width":3.2}));

  /* 交点 */
  const d = x => C(x) - L(x);
  const roots = findRoots(d, 1e-7, XMAX, 900);
  roots.forEach(r=>{
    board.appendChild(mk("circle", {cx:m.X(r), cy:m.Y(C(r)), r:5.5,
      fill:"#dc2626", stroke:"#fff", "stroke-width":2}));
  });

  board.appendChild(mk("circle", {cx:m.X(0), cy:m.Y(0), r:5, fill:"#2563eb"}));
  board.appendChild(mk("text", {x:m.X(0)+10, y:m.Y(0)-10,
    "font-size":12.5, fill:"#2563eb"}, "O"));

  /* 曲线标签 */
  const l1 = mk("text", {x:m.X(1.55), y:m.Y(C(1.55))-16, "font-size":13.5,
    "font-weight":700, fill:"#1d4ed8"});
  l1.appendChild(mk("tspan", {}, "y = e"));
  l1.appendChild(mk("tspan", {"font-size":"10", dy:"-5"}, "x"));
  l1.appendChild(mk("tspan", {dy:"5"}, " − x²/2 − 1"));
  board.appendChild(l1);

  board.appendChild(mk("text", {x:m.X(2.42), y:m.Y(L(2.42))-14,
    "font-size":13.5, "font-weight":700, fill:"#b45309"}, "y = ax"));

  return { roots: roots, hasGap: roots.length > 0 };
}

/* ==================== 步骤 4：证明 ==================== */

/* ==================== 主绘制 ==================== */
function draw(){
  while(board.firstChild) board.removeChild(board.firstChild);
  const a = state.a;
  let info = {};
  switch(state.step){
    case 0: info = drawStep0(a); break;
    case 1: info = drawStep1(a); break;
    case 2: info = drawStep2(a); break;
    case 3: info = drawStep3(a); break;
    case 4: info = drawStep4(a); break;
    case 5: info = drawStep5(a); break;
  }
  return info;
}

  else{
    html = `
      <h2>第 ⑥ 步 · 全部结论</h2>
      <div class="answerbox">
        <div class="lab">（1）a 的取值范围</div>
        <div><b class="hl-g">a ≤ 1</b></div>
        <div class="lab" style="margin-top:6px">（2）(i) 有两个零点时 a 的范围</div>
        <div><b class="hl-g">a &gt; 0</b></div>
        <div class="lab" style="margin-top:6px">（2）(ii) 不等式</div>
        <div><b class="hl-g">3e<sup>x₁</sup> + e<sup>x₂</sup> &gt; 3a</b>　（严格大于）</div>
      </div>
      <p style="margin-top:11px"><b>取等与边界：</b></p>
      <p class="muted">· a = 1 时，(1) 中两边在 x = 0 处同取 0，即 <b>等号只在 x = 0 处成立</b>；x &gt; 0 时
        e<sup>x</sup> − x²/2 − 1 − x = x³/6 + … &gt; 0，仍严格成立，故 a = 1 可取。</p>
      <p class="muted">· a = 0 时 f(x) = e<sup>x</sup> − 3 只有一个零点，(2)(i) 严格排除 a = 0。</p>
      <p class="muted">· (2)(ii) 中，若 x₂ 取到 3 也不用担心：此时 e<sup>x₂</sup> ≥ e³ &gt; 3a 自动成立；
        而在 0 &lt; x₂ &lt; 3 时由 e<sup>t</sup>(3−t) ≤ e² &lt; 9 保证严格不等号。</p>
    `;
  }
  cardEl.innerHTML = html;
}

/* ==================== 交互 ==================== */
let raf = null;
function scheduleRender(){
  if(raf) return;
  raf = requestAnimationFrame(()=>{
    raf = null;
    const info = draw();
    renderCard(info);
  });
}

function applyStep(i){
  state.step = i;
  const conf = STEPS[i];
  state.a = conf.a.def;
  aSlider.min = conf.a.min;
  aSlider.max = conf.a.max;
  aSlider.step = conf.a.step;
  aSlider.value = state.a;
  aOut.textContent = Number(state.a).toFixed(2);
  Array.prototype.forEach.call(stepbarEl.children, (b, k)=>{
    b.classList.toggle("on", k === i);
  });
  const info = draw();
  renderCard(info);
}

function buildStepbar(){
  STEPS.forEach((s, i)=>{
    const b = document.createElement("button");
    b.type = "button";
    b.textContent = s.label;
    b.addEventListener("click", ()=> applyStep(i));
    stepbarEl.appendChild(b);
  });
}

aSlider.addEventListener("input", ()=>{
  state.a = parseFloat(aSlider.value);
  aOut.textContent = state.a.toFixed(2);
  scheduleRender();
});

/* ==================== 启动 ==================== */
buildStepbar();
applyStep(0);
