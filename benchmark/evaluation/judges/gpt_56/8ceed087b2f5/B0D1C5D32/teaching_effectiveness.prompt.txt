  const X_STOP = V0 * V0 / (2 * A); // 108 m

  const RX0 = 70;         // 跑道上 x=0 m 的像素位置
  const RS  = 7.6;        // 每米像素
  const GROUND = 190;     // 地面 y

  const GX0 = 150, GY0 = 560, GTX = 110, GTY = 7.5; // v-t 图坐标变换

  const NS = 'http://www.w3.org/2000/svg';
  const $ = id => document.getElementById(id);

  function ce(tag, attrs, parent){
    const e = document.createElementNS(NS, tag);
    if (attrs) for (const k in attrs) e.setAttribute(k, attrs[k]);
    if (parent) parent.appendChild(e);
    return e;
  }

  /* ---------- 构建跑道刻度 ---------- */
  const ruler = $('ruler');
  for (let m = 0; m <= 110; m += 10){
    const x = RX0 + m * RS;
    const major = (m % 20 === 0);
    ce('line', {
      x1:x, x2:x,
      y1: major ? 181 : 185,
      y2: 190,
      stroke: major ? '#8fa4be' : '#bacada',
      'stroke-width': major ? 1.6 : 1
    }, ruler);
    if (major){
      const t = ce('text', { x:x, y:232, 'text-anchor':'middle', 'class':'tick' }, ruler);
      t.textContent = m;
    }
  }

  /* ---------- 构建 v-t 网格与刻度 ---------- */
  const grid = $('gridLines');
  for (let k = 1; k <= 6; k++){
    const x = GX0 + k * GTX;
    ce('line', { x1:x, y1:560, x2:x, y2:296, stroke:'#e9eff7', 'stroke-width':1 }, grid);
  }
  for (let v = 12; v <= 36; v += 12){
    const y = GY0 - v * GTY;
    ce('line', { x1:150, y1:y, x2:888, y2:y, stroke:'#e9eff7', 'stroke-width':1 }, grid);
  }

  const axisLabels = $('axisLabels');
  const z = ce('text', { x:142, y:566, 'text-anchor':'end', 'class':'tick' }, axisLabels);
  z.textContent = '0';
  for (let k = 1; k <= 6; k++){
    const x = GX0 + k * GTX;
    ce('line', { x1:x, y1:560, x2:x, y2:566, stroke:'#93a6bd', 'stroke-width':1.4 }, axisLabels);
    const t = ce('text', { x:x, y:584, 'text-anchor':'middle', 'class':'tick' }, axisLabels);
    t.textContent = k;
  }
  for (let v = 12; v <= 36; v += 12){
    const y = GY0 - v * GTY;
    ce('line', { x1:144, y1:y, x2:150, y2:y, stroke:'#93a6bd', 'stroke-width':1.4 }, axisLabels);
    const t = ce('text', { x:140, y:y + 4, 'text-anchor':'end', 'class':'tick' }, axisLabels);
    t.textContent = v;
  }

  /* ---------- 元素引用 ---------- */
  const plane     = $('plane');
  const travelBar = $('travelBar');
  const lastSeg   = $('lastSeg');
  const vGroup    = $('vGroup');
  const vLine     = $('vLine');
  const vText     = $('vText');
  const aGroup    = $('aGroup');
  const aLine     = $('aLine');
  const aText     = $('aText');
  const ptr       = $('ptr');
  const ptrDot    = $('ptrDot');
  const areaTotal = $('areaTotal');
  const areaFirst = $('areaFirst');
  const areaLast  = $('areaLast');
  const lblTotal  = $('lblTotal');
  const lblFirst  = $('lblFirst');
  const lblLast   = $('lblLast');
  const stopPt    = $('stopPt');
  const bracket   = $('bracket');
  const marker108 = $('marker108');
  const hudT = $('hudT'), hudV = $('hudV'), hudX = $('hudX');
  const slider = $('timeSlider');
  const sliderLabel = $('sliderLabel');
  const playBtn = $('playBtn');
  const resetBtn = $('resetBtn');
  const noteEl = $('note');
  const stepBtns = Array.prototype.slice.call(document.querySelectorAll('.stepBtn'));

  /* ---------- 状态 ---------- */
  const state = { t:0, playing:false, step:0 };

  /* ---------- 渲染 ---------- */
  function render(){
    const t = Math.max(0, Math.min(T_STOP, state.t));
    const v = Math.max(0, V0 - A * t);
    const x = V0 * t - 0.5 * A * t * t;
    const px = RX0 + x * RS;

    plane.setAttribute('transform', 'translate(' + px.toFixed(2) + ',' + GROUND + ')');
    travelBar.setAttribute('width', Math.max(0, px - RX0).toFixed(2));

    /* 速度箭头 */
    const len = v * 3.3;
    if (len > 7){
      const x1 = px + 50, x2 = x1 + len;
      vGroup.setAttribute('opacity', '1');
      vLine.setAttribute('x1', x1.toFixed(1));
      vLine.setAttribute('x2', x2.toFixed(1));
      vText.setAttribute('x', ((x1 + x2) / 2).toFixed(1));
      vText.textContent = 'v = ' + v.toFixed(1) + ' m/s';
    } else {
      vGroup.setAttribute('opacity', '0');
    }

    /* 加速度箭头 */
    if (v > 0.05){
      aGroup.setAttribute('opacity', '1');
      aLine.setAttribute('x1', (px + 30).toFixed(1));
      aLine.setAttribute('x2', (px - 40).toFixed(1));
      aText.setAttribute('x', (px - 5).toFixed(1));
    } else {
      aGroup.setAttribute('opacity', '0');
    }

    /* 图像指针 */
    const gx = GX0 + t * GTX;
    const gy = GY0 - v * GTY;
    ptr.setAttribute('x1', gx.toFixed(1));
    ptr.setAttribute('x2', gx.toFixed(1));
    ptr.setAttribute('y2', gy.toFixed(1));
    ptrDot.setAttribute('cx', gx.toFixed(1));
    ptrDot.setAttribute('cy', gy.toFixed(1));

    /* 读数 */
    hudT.textContent = t.toFixed(1);
    hudV.textContent = v.toFixed(1);
    hudX.textContent = x.toFixed(1);
    slider.value = t;
    sliderLabel.textContent = 't = ' + t.toFixed(1) + ' s';
  }

  /* ---------- 步骤高亮 ---------- */
  function applyStep(){
    const s = state.step;
    const show = (el, on) => el.setAttribute('opacity', on ? '1' : '0');

    areaTotal.setAttribute('opacity', s === 2 ? '0.45' : '0');
    areaFirst.setAttribute('opacity', s === 3 ? '0.22' : '0');
    areaLast.setAttribute('opacity',  s === 3 ? '0.62' : '0');

    show(lblTotal, s === 2);
    show(lblFirst, s === 3);
    show(lblLast,  s === 3);
    show(stopPt,   s >= 1);
    show(bracket,  s >= 3);
    show(marker108, s >= 2);
    show(lastSeg,   s >= 3);
  }

  /* ---------- 步骤文案 ---------- */
  const NOTES = [
    '<h3>① 读题：把文字变成图</h3>' +
    '<p>已知：<span class="m">v₀ = 36 m/s</span>，加速度大小 <span class="m">a = 6 m/s²</span>，方向与运动相反（匀减速），一直滑到 <span class="m">v = 0</span>。</p>' +
    '<p>换句话说，飞机“<b>每秒慢 6 m/s</b>”。要求：（1）滑行距离；（2）最后 4 s 内的位移。</p>' +
    '<p>拖动下面的时间滑块，看飞机的停位和右边 v–t 图上的点怎么一起变化。</p>',

    '<h3>② 什么时候停下？</h3>' +
    '<p>速度每秒少 6 m/s，从 36 m/s 减到 0，需要 <b>36 ÷ 6 = 6 s</b>。</p>' +
    '<p>写成公式：<b>v = v₀ − a t</b>，令 v = 0 得 <b>0 = 36 − 6t，t = 6 s</b>。</p>' +
    '<p>看右图：蓝色直线与横轴交于 <b>t = 6 s</b> —— 这就是滑行的总时间。</p>',

    '<h3>③ 第（1）问：距离 = 图像下的面积</h3>' +
    '<p>v–t 图线下的面积就是位移。整块蓝色三角形：</p>' +
    '<p><b>S = ½ × 6 s × 36 m/s = 108 m</b></p>' +
    '<p>也可以直接代入：<b>S = v₀² / (2a) = 36² / (2×6) = 108 m</b>。跑道上的橙色虚线就是终点。</p>',

    '<h3>④ 第（2）问：只看最后 4 s</h3>' +
    '<p>停止时刻是 6 s，所以“最后 4 s”就是 <b>t = 2 s → 6 s</b>（图上橙色区域，对应跑道上 60 m → 108 m 那一段）。</p>' +
    '<p>t = 2 s 时速度 v = 36 − 6×2 = 24 m/s，梯形面积：<b>½ × (24 + 0) × 4 = 48 m</b>。</p>' +
    '<p>更巧的看法：把过程<b>倒过来看</b> —— 从静止出发，用 4 s 匀加速到 24 m/s，走过 <span class="o">½ × 6 × 4² = 48 m</span>。</p>' +
    '<p>也可以：108 − 前 2 s 的 60 = <span class="o">48 m</span>，两种算法结果一致。</p>',

    '<h3>⑤ 结论</h3>' +
    '<p>（1）滑行距离 <b>108 m</b>；（2）最后 4 s 内的位移 <b>48 m</b>。</p>' +
    '<p>抓住一条主线：匀减速运动的 v–t 图是一条<b>斜向下的直线</b> —— <b>截距给出停止时刻，斜率的大小就是加速度，图线下的面积就是位移</b>。</p>'
  ];

  /* ---------- 交互 ---------- */
  function setPlaying(p){
    state.playing = p;
    playBtn.textContent = p ? '暂停' : '播放';
  }

  function setStep(s){
    state.step = s;
    stepBtns.forEach(b => b.classList.toggle('active', Number(b.dataset.step) === s));
    noteEl.innerHTML = NOTES[s];

    if (s === 0){
      state.t = 0;
      setPlaying(false);
    } else if (s <= 3){
      state.t = T_STOP;
      setPlaying(false);
    }
    render();
    applyStep();
  }

  playBtn.addEventListener('click', function(){
    if (state.t >= T_STOP - 1e-6) state.t = 0;
    setPlaying(!state.playing);
    render();
  });

  resetBtn.addEventListener('click', function(){
    setPlaying(false);
    state.t = 0;
    render();
  });

  slider.addEventListener('input', function(){
    setPlaying(false);
    state.t = parseFloat(slider.value);
    render();
  });

  stepBtns.forEach(function(b){
    b.addEventListener('click', function(){ setStep(Number(b.dataset.step)); });
  });

  /* ---------- 动画循环 ---------- */
  let last = 0;
  function loop(now){
    if (!last) last = now;
    const dt = Math.min((now - last) / 1000, 0.05);
    last = now;
    if (state.playing){
      state.t += dt;
      if (state.t >= T_STOP){
        state.t = T_STOP;
        setPlaying(false);
      }
      render();
    }
    requestAnimationFrame(loop);
  }

  /* ---------- 初始化 ---------- */
  setStep(0);
  requestAnimationFrame(loop);
})();
</script>
