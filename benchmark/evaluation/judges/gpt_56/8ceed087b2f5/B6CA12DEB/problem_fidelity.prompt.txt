  const sim = { ions: [] };

  function makeIon(type, box){
    const r = ION_R;
    return {
      type,
      x: box.x0 + r + Math.random()*Math.max(1, box.x1-box.x0-2*r),
      y: box.y0 + r + Math.random()*Math.max(1, box.y1-box.y0-2*r),
      vx: (Math.random()-0.5)*28,
      vy: (Math.random()-0.5)*28,
      a: 0, alive:true, r
    };
  }

  const ALL_TYPES = ['Ba','OH','H','SO4','Na','K','Mg'];

  function updateIons(dt, counts, box){
    for (let i=0;i<ALL_TYPES.length;i++){
      const t = ALL_TYPES[i];
      const want = Math.round(counts[t] || 0);
      let aliveN = 0;
      for (let k=0;k<sim.ions.length;k++){
        const p = sim.ions[k];
        if (p.type === t && p.alive) aliveN++;
      }
      if (aliveN < want){
        const add = Math.min(want - aliveN, 3);
        for (let k=0;k<add;k++) sim.ions.push(makeIon(t, box));
      } else if (aliveN > want){
        let ex = aliveN - want;
        for (let k=0;k<sim.ions.length && ex>0;k++){
          const p = sim.ions[k];
          if (p.type === t && p.alive){ p.alive = false; ex--; }
        }
      }
    }

    for (let i=0;i<sim.ions.length;i++){
      const p = sim.ions[i];
      p.a += ((p.alive ? 1 : 0) - p.a) * Math.min(1, dt*8);
      p.x += p.vx*dt;
      p.y += p.vy*dt;
      const r = p.r;
      if (p.x < box.x0+r){ p.x = box.x0+r; p.vx = Math.abs(p.vx); }
      if (p.x > box.x1-r){ p.x = box.x1-r; p.vx = -Math.abs(p.vx); }
      if (p.y < box.y0+r){ p.y = box.y0+r; p.vy = Math.abs(p.vy); }
      if (p.y > box.y1-r){ p.y = box.y1-r; p.vy = -Math.abs(p.vy); }
    }

    const next = [];
    for (let i=0;i<sim.ions.length;i++){
      const p = sim.ions[i];
      if (p.alive || p.a > 0.02) next.push(p);
    }
    sim.ions = next;
  }

  /* 沉淀位置的伪随机固定分布 */
  const SPOTS = (function(){
    let s = 20240517;
    const rnd = function(){ s = (s*1103515245 + 12345) & 0x7fffffff; return s / 0x7fffffff; };
    const arr = [];
    for (let i=0;i<24;i++) arr.push({ fx: rnd(), fy: rnd(), r: 2.1 + rnd()*2.4 });
    return arr;
  })();

  /* ============================================================
     五、绘图：烧杯 + 离子
     ============================================================ */
  function simBox(w, h){
    const bx0 = w*0.10, bx1 = w*0.90;
    const byTop = h*0.20, byBot = h*0.965;
    const liquidTop = byTop + (byBot-byTop)*0.15;
    return {
      bx0:bx0, bx1:bx1, byTop:byTop, byBot:byBot, liquidTop:liquidTop,
      x0: bx0+20, x1: bx1-20,
      y0: liquidTop+20, y1: byBot-28
    };
  }

  function drawSim(dt, counts){
    const w = simSize.w, h = simSize.h;
    if (w < 2 || h < 2) return;
    const ctx = simCtx;
    ctx.clearRect(0,0,w,h);

    const B = simBox(w,h);
    const bx0 = B.bx0, bx1 = B.bx1, byTop = B.byTop, byBot = B.byBot, liquidTop = B.liquidTop;

    /* 烧杯外形 */
    ctx.beginPath();
    ctx.moveTo(bx0, byTop);
    ctx.lineTo(bx0, byBot-13);
    ctx.quadraticCurveTo(bx0, byBot, bx0+13, byBot);
    ctx.lineTo(bx1-13, byBot);
    ctx.quadraticCurveTo(bx1, byBot, bx1, byBot-13);
    ctx.lineTo(bx1, byTop);
    ctx.strokeStyle = '#b3c2d6';
    ctx.lineWidth = 2.5;
    ctx.lineJoin = 'round';
    ctx.stroke();

    /* 液体 */
    ctx.beginPath();
    ctx.moveTo(bx0+1.5, liquidTop);
    ctx.lineTo(bx0+1.5, byBot-13);
    ctx.quadraticCurveTo(bx0+1.5, byBot-1.5, bx0+13, byBot-1.5);
    ctx.lineTo(bx1-13, byBot-1.5);
    ctx.quadraticCurveTo(bx1-1.5, byBot-1.5, bx1-1.5, byBot-13);
    ctx.lineTo(bx1-1.5, liquidTop);
    ctx.closePath();
    const lg = ctx.createLinearGradient(0, liquidTop, 0, byBot);
    lg.addColorStop(0, '#f2f8ff');
    lg.addColorStop(1, '#d9ebfd');
    ctx.fillStyle = lg;
    ctx.fill();

    /* 液面线 */
    ctx.beginPath();
    ctx.moveTo(bx0+2, liquidTop);
    ctx.lineTo(bx1-2, liquidTop);
    ctx.strokeStyle = '#a9cdf0';
    ctx.lineWidth = 2;
    ctx.stroke();

    /* 沉淀 */
    const sc = SCEN[state.scene];
    const amt = sc.precip ? sc.precip(state.x) : 0;
    const nSpot = Math.round(amt * SPOTS.length);
    for (let i=0;i<nSpot;i++){
      const s = SPOTS[i];
      const px = bx0 + 24 + s.fx*(bx1-bx0-48);
      const py = byBot - 24 - s.fy*10;
      ctx.beginPath();
      ctx.arc(px, py, s.r, 0, Math.PI*2);
      ctx.fillStyle = 'rgba(255,255,255,.96)';
      ctx.fill();
      ctx.strokeStyle = 'rgba(126,160,205,.85)';
      ctx.lineWidth = 1;
      ctx.stroke();
    }

    /* 离子 */
    updateIons(dt, counts, B);
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    for (let i=0;i<sim.ions.length;i++){
      const p = sim.ions[i];
      const st = ION_STYLE[p.type];
      if (!st) continue;
      const al = Math.max(0, Math.min(1, p.a));
      if (al < 0.03) continue;
      const rr = p.r * (0.35 + 0.65*al);
      ctx.globalAlpha = al;
      ctx.beginPath();
      ctx.arc(p.x, p.y, rr, 0, Math.PI*2);
      ctx.fillStyle = st.fill;
      ctx.fill();
      if (al > 0.55 && rr > 8){
        ctx.fillStyle = '#fff';
        ctx.font = '600 9px system-ui,"PingFang SC",sans-serif';
        ctx.fillText(st.label, p.x, p.y+0.5);
      }
      ctx.globalAlpha = 1;
    }

    /* 滴管 */
    const tx = w*0.5;
    const tubeH = Math.max(18, byTop - 14);
    ctx.fillStyle = 'rgba(160,182,210,.45)';
    ctx.fillRect(tx-8, byTop - tubeH - 8, 16, tubeH + 8);
    ctx.fillStyle = 'rgba(160,182,210,.85)';
    ctx.fillRect(tx-14, byTop - tubeH - 16, 28, 10);

    if (state.x > 0.003){
      const t = (performance.now()/650) % 1;
      const dy = (byTop - tubeH) + t * (tubeH + 20);
      ctx.beginPath();
      ctx.ellipse(tx, dy, 3.6, 5, 0, 0, Math.PI*2);
      ctx.fillStyle = 'rgba(120,170,235,.85)';
      ctx.fill();
    }
  }

  /* ============================================================
     六、绘图：曲线
     ============================================================ */
  const PAD = { l:50, r:16, t:16, b:38 };

  function chartGeom(){
    const w = chartSize.w, h = chartSize.h;
    const px0 = PAD.l, px1 = w - PAD.r, py0 = h - PAD.b, py1 = PAD.t;
    return {
      w:w, h:h, px0:px0, px1:px1, py0:py0, py1:py1,
      X: function(v){ return px0 + (v/3)*(px1-px0); },
      Y: function(v){ return py0 - (v/YMAX)*(py0-py1); }
    };
  }

  function plotCurve(ctx, G, fn, color, width, dash){
    ctx.save();
    ctx.setLineDash(dash || []);
    ctx.beginPath();
    ctx.strokeStyle = color;
    ctx.lineWidth = width;
    ctx.lineJoin = 'round';
    ctx.lineCap = 'round';
    for (let i=0;i<=200;i++){
      const xv = i/200*3;
      let yv = fn(xv);
      if (!isFinite(yv)) yv = 0;
      yv = Math.max(0, Math.min(YMAX, yv));
      const px = G.X(xv), py = G.Y(yv);
      if (i===0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
    }
    ctx.stroke();
    ctx.restore();
  }

  function drawChart(){
    const w = chartSize.w, h = chartSize.h;
    if (w < 2 || h < 2) return;
    const ctx = chartCtx;
    ctx.clearRect(0,0,w,h);

    const G = chartGeom();
    const px0 = G.px0, px1 = G.px1, py0 = G.py0, py1 = G.py1;

    /* 网格 */
    ctx.strokeStyle = '#eef3f9';
    ctx.lineWidth = 1;
    for (let v=0; v<=600; v+=150){
      ctx.beginPath();
      ctx.moveTo(px0, G.Y(v));
      ctx.lineTo(px1, G.Y(v));
      ctx.stroke();
    }

    /* 坐标轴 */
    ctx.strokeStyle = '#8496ad';
    ctx.lineWidth = 1.6;
    ctx.beginPath();
    ctx.moveTo(px0, py1-10);
    ctx.lineTo(px0, py0);
    ctx.lineTo(px1+10, py0);
    ctx.stroke();

    ctx.fillStyle = '#8496ad';
    ctx.beginPath();
    ctx.moveTo(px0-5, py0+9);
    ctx.lineTo(px0+5, py0+9);
    ctx.lineTo(px0, py0-1);
    ctx.closePath();
    ctx.fill();

    ctx.beginPath();
    ctx.moveTo(px1+2, py0-5);
    ctx.lineTo(px1+2, py0+5);
    ctx.lineTo(px1+12, py0);
    ctx.closePath();
    ctx.fill();

    /* 轴标题 */
    ctx.save();
    ctx.translate(14, (py0+py1)/2);
    ctx.rotate(-Math.PI/2);
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillStyle = '#5b6779';
    ctx.font = '12px system-ui,"PingFang SC",sans-serif';
    ctx.fillText('导电能力', 0, 0);
    ctx.restore();

    ctx.fillStyle = '#5b6779';
    ctx.font = '12px system-ui,"PingFang SC",sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    ctx.fillText('时间 / 滴加量', (px0+px1)/2, py0+20);

    /* 原题曲线（对照） */
    if (state.scene !== 'original'){
      plotCurve(ctx, G, SCEN.original.kappa, '#c8d3e2', 2, [5,5]);
    }

    /* 当前曲线 */
    plotCurve(ctx, G, SCEN[state.scene].kappa, '#2b6cff', 2.8, null);

    /* 原图的 a 点 */
    if (state.scene === 'original'){
      const ax = G.X(1);
      ctx.save();
      ctx.setLineDash([3,4]);
      ctx.strokeStyle = '#c2cfe0';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(ax, py0);
      ctx.lineTo(ax, G.Y(SCEN.original.kappa(1)) + 2);
      ctx.stroke();
      ctx.restore();

      ctx.fillStyle = '#8b9ab0';
      ctx.font = 'italic 14px Georgia,serif';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'top';
      ctx.fillText('a', ax, py0 + 5);
    }

    /* 与画布对应的当前点 */
    const cur = SCEN[state.scene].kappa(state.x);
    const cx = G.X(state.x), cy = G.Y(Math.max(0, Math.min(YMAX, cur)));

    ctx.save();
    ctx.setLineDash([4,4]);
    ctx.strokeStyle = 'rgba(43,108,255,.32)';
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.moveTo(cx, py0);
    ctx.lineTo(cx, cy);
    ctx.stroke();
    ctx.restore();

    ctx.beginPath();
    ctx.arc(cx, cy, 6.2, 0, Math.PI*2);
    ctx.fillStyle = '#2b6cff';
    ctx.fill();
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 2.2;
    ctx.stroke();

    /* 曲线关键点标注 */
    ctx.font = '600 12.5px system-ui,"PingFang SC",sans-serif';
    ctx.fillStyle = '#5b6779';
    ctx.textAlign = 'left';
    ctx.textBaseline = 'bottom';
    ctx.fillText('A', G.X(0.16)+2, G.Y(SCEN[state.scene].kappa(0.16)) - 8);

    const lowX = state.scene === 'khso4' ? 2 : (state.scene === 'naoh' ? 0.5 : 1);
    const lowY = SCEN[state.scene].kappa(lowX);
    ctx.textAlign = 'center';
    ctx.fillText('B', G.X(lowX), G.Y(lowY) - 12);

    ctx.textAlign = 'left';
    ctx.textBaseline = 'bottom';
    ctx.fillText('C', G.X(2.55)+2, G.Y(SCEN[state.scene].kappa(2.55)) - 10);
  }

  function syncLegend(){
    const sc = SCEN[state.scene];
    let html = '';
    for (let i=0;i<sc.legend.length;i++){
      const t = sc.legend[i];
      html += '<span><i style="background:'+ION_STYLE[t].fill+'"></i>'+ION_TEXT[t]+'</span>';
    }
    if (sc.precip && sc.precip(1) > 0){
      html += '<span><i style="background:#fff;border:1px solid #9fb0c6"></i>沉淀</span>';
    }
    legendEl.innerHTML = html;
    eqLine.innerHTML = sc.eq.replace(/→/g, '<span class="arrow">→</span>');
  }

  let lastStatus = -999;
  function syncUI(){
    const sc = SCEN[state.scene];
    const k = sc.kappa(state.x);
    const pct = Math.max(0, Math.min(100, k / 527 * 100));

    meterFill.style.width = pct.toFixed(1) + '%';
    meterVal.textContent = Math.round(pct) + '%';
    xVal.textContent = state.x.toFixed(2);

    const note = sc.note(state.x);
    if (note !== statusEl.textContent) statusEl.textContent = note;

    const nearLow = Math.abs(state.x - sc.eqX) < 0.06;
    statusEl.classList.toggle('hot', nearLow);

    if (slider.value !== String(state.x)){
      // 避免与用户拖动冲突：只在播放时由外部同步
    }
  }

  /* ============================================================
     八、渲染循环
     ============================================================ */
  let lastT = performance.now();

  function loop(now){
    let dt = (now - lastT) / 1000;
    lastT = now;
    if (dt > 0.06) dt = 0.06;

    if (state.playing){
      state.x += dt * 0.55;
      if (state.x >= 3){ state.x = 3; state.playing = false; updatePlayBtn(); }
      slider.value = state.x;
    }

    const counts = SCEN[state.scene].counts(state.x);
    drawSim(dt, counts);
    drawChart();
    syncUI();

    requestAnimationFrame(loop);
  }

  /* ============================================================
     九、尺寸
     ============================================================ */
  function resize(){
    const dpr = Math.min(window.devicePixelRatio || 1, 2);

    let r = simCanvas.getBoundingClientRect();
    simSize.w = Math.max(1, Math.round(r.width));
    simSize.h = Math.max(1, Math.round(r.height));
    simCanvas.width  = Math.round(simSize.w * dpr);
    simCanvas.height = Math.round(simSize.h * dpr);
    simCtx.setTransform(dpr,0,0,dpr,0,0);

    r = chartCanvas.getBoundingClientRect();
    chartSize.w = Math.max(1, Math.round(r.width));
    chartSize.h = Math.max(1, Math.round(r.height));
    chartCanvas.width  = Math.round(chartSize.w * dpr);
    chartCanvas.height = Math.round(chartSize.h * dpr);
    chartCtx.setTransform(dpr,0,0,dpr,0,0);
  }

  window.addEventListener('resize', resize);

     十、交互
     ============================================================ */

    state.x = Math.max(0, Math.min(3, v));
    slider.value = state.x;

    state.playing = false;
    updatePlayBtn();
    setX(parseFloat(slider.value));

    playBtn.textContent = state.playing ? '暂停' : '播放';
  }

  playBtn.addEventListener('click', function(){
    if (state.playing){
      state.playing = false;
    } else {
      if (state.x >= 2.995) setX(0);
      state.playing = true;
    }
    updatePlayBtn();
  });

  /* 在曲线上横向拖动 */
  let dragging = false;

  function xFromEvent(e){
    const rect = chartCanvas.getBoundingClientRect();
    const G = chartGeom();
    const local = e.clientX - rect.left;
    const v = (local - G.px0) / (G.px1 - G.px0) * 3;
    return v;
  }

  chartCanvas.addEventListener('pointerdown', function(e){
    dragging = true;
    state.playing = false;
    updatePlayBtn();
    try { chartCanvas.setPointerCapture(e.pointerId); } catch(err){}
    setX(xFromEvent(e));
  });

  chartCanvas.addEventListener('pointermove', function(e){
    if (!dragging) return;
    setX(xFromEvent(e));
  });

  chartCanvas.addEventListener('pointerup', function(e){
    dragging = false;
    try { chartCanvas.releasePointerCapture(e.pointerId); } catch(err){}
  });

  chartCanvas.addEventListener('pointercancel', function(){ dragging = false; });

  /* 选项卡片 */
  let activeOpt = null;

  function buildOptions(){
    let html = '';
    for (let i=0;i<OPTIONS.length;i++){
      const o = OPTIONS[i];
      html += '<button class="opt" data-id="'+o.id+'" type="button">'
            +   '<span class="ltr">'+o.id+'</span>'
            +   '<span class="txt">'+o.text+'</span>'
            + '</button>';
    }
    optGrid.innerHTML = html;

    const btns = optGrid.querySelectorAll('.opt');
    for (let i=0;i<btns.length;i++){
      btns[i].addEventListener('click', function(){
        const id = this.getAttribute('data-id');
        selectOption(id);
      });
    }
  }

  function selectOption(id){
    const o = OPTIONS.filter(function(x){ return x.id === id; })[0];
    if (!o) return;

    activeOpt = id;

    /* 卡片状态 */
    const btns = optGrid.querySelectorAll('.opt');
    for (let i=0;i<btns.length;i++){
      const b = btns[i];
      const on = b.getAttribute('data-id') === id;
      b.classList.toggle('on', on);
      const old = b.querySelector('.res');
      if (old) old.remove();
      if (on){
        const tag = document.createElement('span');
        tag.className = 'res ' + (o.ok ? 'ok' : 'no');
        tag.textContent = o.ok ? '成立' : '不成立';
        b.appendChild(tag);
      }
    }

    /* 反馈 */
    feedback.className = 'feedback show ' + (o.ok ? 'ok' : 'no');
    feedback.innerHTML = '<b>' + (o.ok ? '正确。' : '错误。') + '</b>' + o.fb;

    /* 曲线场景 */
    state.playing = false;
    updatePlayBtn();

    if (o.scen){
      state.scene = o.scen;
      setX(1.5);
    } else {
      state.scene = 'original';
      setX(1.0);
    }

    syncLegend();
  }

  /* ============================================================
     十一、启动
     ============================================================ */
  buildOptions();
  syncLegend();
  resize();
  syncLegend();
  requestAnimationFrame(function(t){
    lastT = t;
    requestAnimationFrame(loop);
  });

  /* 首次尺寸可能在字体加载后变化 */
  setTimeout(resize, 60);
  setTimeout(resize, 300);

})();
