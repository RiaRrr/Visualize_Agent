    return function () {
      var t = a += 0x6D2B79F5;
      t = Math.imul(t ^ t >>> 15, t | 1);
      t ^= t + Math.imul(t ^ t >>> 7, t | 61);
      return ((t ^ t >>> 14) >>> 0) / 4294967296;
    };
  }

  function conductivity(exp, t) {
    t = Math.max(0, Math.min(2, t));
    if (exp === 'h2so4') {
      if (t <= 1) return (1 - t) / (1 + 0.5 * t);
      return (t - 1) / (1 + 0.5 * t);
    }
    if (exp === 'mgso4') {
      if (t <= 1) return (1 - t) / (1 + 0.5 * t);
      return 0.9 * (t - 1) / (1 + 0.5 * t);
    }
    if (exp === 'naoh') {
      if (t <= 1) return (1 - 0.2 * t) / (1 + 0.1 * t);
      return (0.8 + 0.3 * (t - 1)) / (1 + 0.1 * t);
    }
    if (exp === 'khso4') {
      if (t <= 1) return (1 - 0.5 * t) / (1 + 0.25 * t);
      return (0.5 + 0.5 * (t - 1)) / (1 + 0.25 * t);
    }
    return 0;
  }

  function getState(exp, t) {
    let counts = { ba: 0, oh: 0, h: 0, so4: 0, k: 0, na: 0, mg: 0 };
    let precip = { baso4: 0, mgoh2: 0 };
    t = Math.max(0, Math.min(2, t));

    if (exp === 'h2so4') {
      if (t <= 1) {
        const n = Math.min(8, Math.round(8 * t));
        precip.baso4 = n;
        counts.ba = 8 - n;
        counts.oh = 2 * (8 - n);
      } else {
        precip.baso4 = 8;
        const excess = Math.round(8 * (t - 1));
        counts.h = 2 * excess;
        counts.so4 = excess;
      }
    } else if (exp === 'naoh') {
      counts.na = 8;
      if (t <= 1) {
        counts.oh = Math.max(0, Math.round(8 * (1 - t)));
        counts.so4 = Math.round(4 * t);
      } else {
        counts.so4 = Math.round(4 * t);
        counts.h = Math.round(8 * (t - 1));
      }
    } else if (exp === 'khso4') {
      const n = 8 * t;
      counts.k = Math.round(n);
      if (n <= 8) {
        counts.ba = Math.round(8 - n);
        counts.oh = Math.round(16 - n);
        precip.baso4 = Math.round(n);
      } else {
        counts.oh = Math.max(0, Math.round(16 - n));
        counts.so4 = Math.round(n - 8);
        precip.baso4 = 8;
      }
    } else if (exp === 'mgso4') {
      if (t <= 1) {
        const n = Math.min(8, Math.round(8 * t));
        precip.baso4 = n;
        precip.mgoh2 = n;

        precip.mgoh2 = 8;
        const excess = Math.round(8 * (t - 1));
        counts.mg = excess;
        counts.so4 = excess;
      }
    }
    return { counts, precip };
  }

    label.setAttribute('fill', minV < 0.05 ? '#16a34a' : '#475569');
    const guide = document.getElementById('miniMinGuide');
    guide.setAttribute('x1', bx);
    guide.setAttribute('y1', by);
    guide.setAttribute('x2', labelX - 2);
    guide.setAttribute('y2', labelY + 3);
    guide.setAttribute('stroke', minV < 0.05 ? '#16a34a' : '#94a3b8');
    const spec = experimentSpecs[exp];
    document.getElementById('miniExpLabel').textContent = '底液：' + spec.base + ' · 滴加：' + spec.titrant;
    document.getElementById('miniMinValue').textContent = minV < 0.05 ? '接近 0' : '不为 0';
    document.getElementById('miniResidual').textContent = exp === 'naoh' ? '残留 Na⁺' : (exp === 'khso4' ? '残留 K⁺' : '残留离子少');
  }

  function update(skipNarrative) {
    const exp = state.exp, t = state.t;
    const dose = document.getElementById('dose');
    dose.value = t;
    document.getElementById('doseValue').textContent = t.toFixed(2);

    document.querySelectorAll('.exp-btn').forEach(function (btn) {
      btn.classList.toggle('active', btn.dataset.exp === exp);
    });

    if (exp === 'naoh') state.highlightResidual = 'na';
    else if (exp === 'khso4') state.highlightResidual = 'k';
    else state.highlightResidual = null;

    const spec = experimentSpecs[exp];
    document.getElementById('chartExpLabel').textContent = '底液：' + spec.base + ' · 滴加：' + spec.titrant;
    document.getElementById('beakerBaseLabel').textContent = '底液：' + spec.base;
    document.getElementById('beakerTitrantLabel').textContent = '滴加：' + spec.titrant;
    const droplet = document.getElementById('droplet');
    droplet.setAttribute('fill', titrantColors[spec.titrantKey]);
    if (state.playing && t < 1.99) {
      droplet.style.display = '';
      droplet.style.animation = '';
      droplet.style.animationPlayState = 'running';
      droplet.style.opacity = '';
    } else if (t > 0.001 && t < 1.99) {
      droplet.style.display = '';
      droplet.style.animation = 'none';
      droplet.style.opacity = '0.85';
    } else {
      droplet.style.display = 'none';
      droplet.style.animation = '';
      droplet.style.animationPlayState = 'paused';
    }

    drawCurve();
    drawMini(exp);

    const stateData = getState(exp, t);
    renderBeaker(stateData);
    renderCounts(stateData);

    const v = conductivity(exp, t);
    document.getElementById('conductValue').textContent = v.toFixed(2);
    document.getElementById('conductBar').style.width = Math.min(100, Math.max(0, v * 100)) + '%';
    document.getElementById('statusLine').textContent = getStatusText(exp, t);
    if (!skipNarrative) syncNarrative(exp, t);
  }

  function stopPlay() {
    state.playing = false;
    if (state.rafId) cancelAnimationFrame(state.rafId);
    state.rafId = null;
    document.getElementById('playBtn').textContent = '播放';
  }

  function togglePlay() {
    if (state.playing) {
      stopPlay();
      return;
    }
    state.playing = true;
    document.getElementById('playBtn').textContent = '暂停';
    if (state.t >= 1.99) state.t = 0;

    let last = performance.now();
    function frame(now) {
      if (!state.playing) return;
      const dt = Math.min(60, now - last);
      last = now;
      state.t = Math.min(2, state.t + dt * 0.00035);
      update();
      if (state.t >= 1.999) {
        stopPlay();
      } else {
        state.rafId = requestAnimationFrame(frame);
      }
    }
    state.rafId = requestAnimationFrame(frame);
  }

    2: { title: '建立表示', desc: '横轴看滴加量，纵轴看相对导电能力；曲线与烧杯中的离子数联动。', exp: 'h2so4', t: 0.4 },
    3: { title: 'AB 段：离子被消耗', desc: 'Ba²⁺ + SO₄²⁻ → BaSO₄↓；H⁺ + OH⁻ → H₂O。自由离子减少，导电能力下降。', exp: 'h2so4', t: 0.6 },
    4: { title: 'B 点：恰好完全反应', desc: '溶液中离子浓度接近 0，导电能力最低。', exp: 'h2so4', t: 1 },
    5: { title: 'BC 段：硫酸过量', desc: '过量的 H₂SO₄ 电离出 H⁺ 和 SO₄²⁻，导电能力回升。', exp: 'h2so4', t: 1.5 },
    6: { title: '对比选项', desc: '换用不同试剂，观察曲线最低点能否接近 0。点击下方选项验证。', exp: 'h2so4', t: 1 }
  };

  function setStepActive(n) {
    document.querySelectorAll('.step-btn').forEach(function (btn) {
      btn.classList.toggle('active', parseInt(btn.dataset.step, 10) === n);
    });
  }

  function setStep(n) {
    stopPlay();
    stepIndex = n;
    setStepActive(n);
    clearOptionFeedback();
    const d = stepData[n];
    state.exp = d.exp;
    state.t = d.t;
    document.getElementById('stepTitle').textContent = d.title;
    document.getElementById('stepDesc').textContent = d.desc;
    update(true);
  }

  function switchExperiment(exp) {
    stopPlay();
    clearOptionFeedback();
    state.exp = exp;
    state.t = 1;
    state.highlightResidual = exp === 'naoh' ? 'na' : (exp === 'khso4' ? 'k' : null);
    stepIndex = 6;
    setStepActive(6);
    const spec = experimentSpecs[exp];
    document.getElementById('stepTitle').textContent = '对比选项';
    document.getElementById('stepDesc').textContent = '当前：底液 ' + spec.base + '，滴加 ' + spec.titrant + '；最低点是否接近 0，看蓝色曲线与横轴的距离。' + (state.highlightResidual ? '图中高亮的是残留在溶液中的离子。' : '恰好反应时离子浓度接近 0。');
    update(true);
  }

  const steps = [
    { n: 1, short: '题设' },
    { n: 2, short: '表示' },
    { n: 3, short: 'AB段' },
    { n: 4, short: 'B点' },
    { n: 5, short: 'BC段' },
    { n: 6, short: '对比' }
  ];

  function initStepNav() {
    const nav = document.getElementById('stepNav');
    nav.innerHTML = steps.map(function (s) {
      return '<button type="button" class="step-btn" data-step="' + s.n + '"><b>' + s.n + '</b>' + s.short + '</button>';
    }).join('');
    nav.querySelectorAll('.step-btn').forEach(function (btn) {
      btn.addEventListener('click', function () {
        setStep(parseInt(btn.dataset.step, 10));
      });
    });
  }

  function handleOption(opt) {
    clearOptionFeedback();
    const btn = document.querySelector('.option[data-opt="' + opt + '"]');
    const feedback = document.getElementById('optionFeedback');
    const conclusion = document.getElementById('finalConclusion');

    if (opt === 'A') {
      stopPlay();
      state.exp = 'h2so4';
      state.t = 1;
      stepIndex = 4;
      setStepActive(4);
      document.getElementById('stepTitle').textContent = 'B 点：恰好完全反应';
      document.getElementById('stepDesc').textContent = '导电弱是因为溶液中自由离子少。BaSO₄ 难溶，但它是强电解质。';
      update(true);
      btn.classList.remove('correct');
      btn.classList.add('wrong');
      feedback.innerHTML = '<b>A 不正确：</b>BaSO₄ 难溶，但溶解的少量仍完全电离，是强电解质。导电能力弱是因为溶液中自由离子少。';
      feedback.className = 'feedback wrong';
      highlightPrecip = true;
      renderBeaker(getState(state.exp, state.t));
      setTimeout(function () {
        highlightPrecip = false;
        renderBeaker(getState(state.exp, state.t));
      }, 1600);
    } else if (opt === 'B') {
      switchExperiment('naoh');
      btn.classList.remove('correct');
      btn.classList.add('wrong');
      feedback.innerHTML = '<b>B 不正确：</b>Na⁺ 和 SO₄²⁻ 始终留在溶液中，因此最低点明显大于 0，不能降到接近 0。';
      feedback.className = 'feedback wrong';
    } else if (opt === 'C') {
      switchExperiment('khso4');
      btn.classList.remove('correct');
      btn.classList.add('wrong');
      feedback.innerHTML = '<b>C 不正确：</b>K⁺ 留在溶液中，因此最低点明显大于 0，不能降到接近 0。';
      feedback.className = 'feedback wrong';
    } else if (opt === 'D') {
      switchExperiment('mgso4');
      btn.classList.remove('wrong');
      btn.classList.add('correct');
      feedback.innerHTML = '<b>D 正确：</b>MgSO₄ 与 Ba(OH)₂ 生成 BaSO₄ 和 Mg(OH)₂ 两种沉淀；恰好反应时离子浓度接近 0，过量后离子增多，曲线先降后升。';
      feedback.className = 'feedback correct';
      conclusion.style.display = 'block';
    }
  }

      });
    });
    document.querySelectorAll('.option').forEach(function (btn) {
      btn.addEventListener('click', function () {
        handleOption(btn.dataset.opt);
      });
    });
    setStep(1);
  });
